\begin{objectivebox}
\textbf{Learning Targets}
\begin{enumerate}
  \item I can state $\mu_{\hat{p}} = p$ and $\sigma_{\hat{p}} = \sqrt{p(1-p)/n}$.
  \item I can check the conditions for using the normal model for $\hat{p}$.
  \item I can find probabilities involving $\hat{p}$ using the normal approximation.
\end{enumerate}
\end{objectivebox}

\begin{vocabbox}
\begin{description}
  \item[\termblanklong{Sampling distribution of $\hat{p}$}] the distribution of all possible values of \blankans{0.5in}{$\hat{p}$} from all possible samples of size \blankans{0.3in}{$n$}.
  \item[\termblanklong{Large Counts condition}] $np \ge$ \blankans{0.4in}{10} \textbf{and} $n(1-p) \ge$ \blankans{0.4in}{10}; ensures the normal model is appropriate.
  \item[\termblanklong{10\% condition}] $n \le$ \blankans{0.4in}{10}\% of the population; ensures approximate independence when sampling without replacement.
\end{description}
\end{vocabbox}

\begin{notesbox}
\textbf{Sampling Distribution of $\hat{p}$} (UNC-3.K, L, M)

\medskip
\begin{tabular}{@{} p{0.2\linewidth} p{0.7\linewidth} @{}}
\textbf{Center:} & $\mu_{\hat{p}} = $ \blankans{1.0in}{$p$} \\[10pt]
\textbf{Spread:} & $\sigma_{\hat{p}} = $ \TallMath{\sqrt{\dfrac{\blankans{1.5in}{$p(1-p)$}}{\blankans{0.5in}{$n$}}}} \\[14pt]
\textbf{Shape:} & Approximately normal when \blankans{1.2in}{$np \ge 10$} and \blankans{1.2in}{$n(1-p) \ge 10$} \\
\end{tabular}
\end{notesbox}

\begin{notesbox}
\textbf{Conditions for Using the Normal Model for $\hat{p}$}

Check all three before computing any probability:
\begin{enumerate}
  \item \textbf{Random:} \blankans{2.5in}{random sample or experiment}
  \item \textbf{10\% condition (Independence):} $n \le 0.10 \cdot N$ where $N$ = \blankans{1.5in}{population size}
  \item \textbf{Large Counts:} $np \ge $ \blankans{0.4in}{10} \textbf{and} $n(1-p) \ge$ \blankans{0.4in}{10}
\end{enumerate}
\end{notesbox}

\begin{notesbox}
\textbf{Worked Example.} In a large city, 62\% of adults have a library card ($p = 0.62$).
A random sample of $n = 150$ adults is selected.

\medskip
\textit{Check conditions:}
\begin{enumerate}
  \item Random: \ansline{stated --- a random sample of 150 adults.}
  \item 10\%: \ansline{$150 \le 10\%$ of all adults in a large city --- yes.}
  \item Large Counts: $np = $ \blankans{0.8in}{93}; $n(1-p) = $ \blankans{0.8in}{57} Both $\ge 10$?~\blankans{0.3in}{Yes}
\end{enumerate}

\medskip
$\mu_{\hat{p}} = $ \blankans{0.6in}{0.62} \hspace{1cm} $\sigma_{\hat{p}} = $ \blankans{1.5in}{$\approx 0.03965$}

\medskip
$P(\hat{p} > 0.65) = $ \ansline{$z = (0.65-0.62)/0.03965 \approx 0.757$}\\\ansline{$P(Z > 0.757) \approx 1 - 0.7754 = 0.2246 \approx 22.5\%$}
\end{notesbox}

\begin{practicebox}
\textbf{Quick Check}

In a large school, 40\% of students take the bus ($p = 0.40$). A random sample of
$n = 80$ students is selected.
\begin{enumerate}
  \item Check all three conditions.
        \ansline{Random: stated.}\\\ansline{10\%: $80 \le 10\%$ of a large school's enrollment --- yes.}\\\ansline{Large Counts: $np = 32 \ge 10$; $n(1-p) = 48 \ge 10$. All conditions met.}
  \item Find $\mu_{\hat{p}}$ and $\sigma_{\hat{p}}$.
        \workans{0.5in}{$\mu_{\hat{p}} = 0.40$; \quad $\sigma_{\hat{p}} = \sqrt{0.40(0.60)/80} = \sqrt{0.003} \approx 0.05477$}
  \item Find $P(\hat{p} \ge 0.50)$.
        \workans{0.8in}{$z = (0.50 - 0.40)/0.05477 \approx 1.826$\\ $P(\hat{p} \ge 0.50) \approx 1 - 0.9660 = 0.0340$. About 3.4\%.}
\end{enumerate}
\end{practicebox}
