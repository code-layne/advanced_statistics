  \boxguard[22]
  \item High school athletes, $p = 0.38$, $n = 180$.
        \begin{enumerate}[label=\alph*.]
          \item \ans{Random: stated. 10\%: $180 \le 10\%$ of high school students nationally --- yes. Large Counts: $np = 180(0.38) = 68.4 \ge 10$; $n(1-p) = 180(0.62) = 111.6 \ge 10$. All conditions met.}\gradedkeyline{\gradedblank[id=hw1a, type=rubric, points=3, display={Random sample; $180 \le 10\%$ of all high school students; $np = 68.4 \ge 10$ and $n(1-p) = 111.6 \ge 10$. All met.}]{\linewidth+6pt}{\gradedlinesheight{2}}}
          \item \ans{$\mu_{\hat{p}} = 0.38$; $\sigma_{\hat{p}} = \sqrt{0.38(0.62)/180} \approx 0.03619$.}\gradedkey{$\mu_{\hat{p}} = $~\gradedblank[id=hw1b1, type=short, points=1, answer=0.38, check=numeric, tolerance=0]{0.9in}{0.28in}\quad $\sigma_{\hat{p}} = $~\gradedblank[id=hw1b2, type=short, points=1, answer=0.0362, check=numeric, tolerance=0.0005]{0.9in}{0.28in}}

          \boxguard[17]
          \item \ans{$z = \text{invNorm}(0.05) \approx -1.645$; $c = 0.38 + (-1.645)(0.03619) \approx 0.320$.
