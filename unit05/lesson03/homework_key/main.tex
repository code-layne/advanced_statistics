\begin{extensionbox}
\textbf{Extension.}
\ans{Need $P(|\bar{x}-50| < 2) \ge 0.95$, i.e., $2/\sigma_{\bar{x}} \ge 1.96$. So $\sigma_{\bar{x}} \le 2/1.96 \approx 1.020$. Then $20/\sqrt{n} \le 1.020 \Rightarrow \sqrt{n} \ge 19.61 \Rightarrow n \ge 385$. Need at least $n = 385$.}
\end{extensionbox}
