\begin{objectivebox}
\textbf{Learning Targets}
\begin{enumerate}
  \item I can state the center and spread of the sampling distribution of $\bar{x}_1 - \bar{x}_2$.
  \item I can check conditions for using the normal model.
  \item I can compute probabilities for $\bar{x}_1 - \bar{x}_2$.
\end{enumerate}
\end{objectivebox}

\begin{notesbox}
\textbf{Sampling Distribution of $\bar{x}_1 - \bar{x}_2$} (UNC-3.T, U, V)

\medskip
\begin{tabular}{@{} p{0.2\linewidth} p{0.72\linewidth} @{}}
\textbf{Center:} & $\mu_{\bar{x}_1-\bar{x}_2} = $ \blankans{1.5in}{$\mu_1 - \mu_2$} \\[12pt]
\textbf{Spread:} & $\sigma_{\bar{x}_1-\bar{x}_2} = $ \TallMath{\sqrt{\dfrac{\blankans{0.5in}{$\sigma_1$}^2}{\blankans{0.4in}{$n_1$}} + \dfrac{\blankans{0.5in}{$\sigma_2$}^2}{\blankans{0.4in}{$n_2$}}}} \\[16pt]
\textbf{Shape:} & Normal/approximately normal when both groups satisfy the \blankans{2.0in}{Normal/Large Sample} condition. \\
\end{tabular}
\end{notesbox}

\begin{notesbox}
\textbf{Conditions for $\bar{x}_1 - \bar{x}_2$}

\begin{enumerate}
  \item \textbf{Random (both groups):} each sample is a random sample.
  \item \textbf{10\% condition (both groups):} $n_1 \le 0.10 N_1$ and $n_2 \le 0.10 N_2$.
  \item \textbf{Independence between groups:} the two samples are \blankans{1.5in}{independent}.
  \item \textbf{Normal/Large Sample (each group):} each population is normal OR each $n \ge$ \blankans{0.4in}{30}.
\end{enumerate}
\end{notesbox}

\begin{notesbox}
\textbf{Worked Example.} Sleep hours: Women $\mu_1 = 7.1$, $\sigma_1 = 1.2$, $n_1 = 50$.
Men: $\mu_2 = 6.8$, $\sigma_2 = 1.4$, $n_2 = 50$.

\medskip
\textit{Check conditions:} \ansline{Random: both stated.}\\\ansline{10\%: both $\le 10\%$ of all women/men. \quad Independent: separate samples.}\\\ansline{Normal/Large: both $n = 50 \ge 30$. All conditions met.}

$\mu_{\bar{x}_1-\bar{x}_2} = $ \blankans{0.5in}{0.3 hr}

$\sigma_{\bar{x}_1-\bar{x}_2} = $ \blankans{2.5in}{$\sqrt{1.2^2/50 + 1.4^2/50} \approx 0.2608$ hr}

$P(\bar{x}_1 - \bar{x}_2 > 1.0) = $ \ansline{$z = (1.0-0.3)/0.2608 \approx 2.684$}\\\ansline{$P \approx 1 - 0.9964 = 0.0036 \approx 0.4\%$}
\end{notesbox}

\begin{practicebox}
\textbf{Quick Check}

Brand A batteries: $\mu_1 = 480$ hr, $\sigma_1 = 60$, $n_1 = 36$.
Brand B batteries: $\mu_2 = 450$ hr, $\sigma_2 = 80$, $n_2 = 49$.

\begin{enumerate}
  \item Find $\mu_{\bar{x}_1-\bar{x}_2}$ and $\sigma_{\bar{x}_1-\bar{x}_2}$.
        \workans{0.6in}{$\mu = 480 - 450 = 30$ hr\\ $\sigma = \sqrt{60^2/36 + 80^2/49} = \sqrt{100 + 130.61} = \sqrt{230.61} \approx 15.19$ hr}
  \item Find $P(\bar{x}_1 - \bar{x}_2 > 40)$.
        \workans{0.8in}{$z = (40-30)/15.19 \approx 0.658$\\ $P(\bar{x}_1 - \bar{x}_2 > 40) \approx 1 - 0.7449 = 0.2551 \approx 25.5\%$}
\end{enumerate}
\end{practicebox}
