\begin{tcolorbox}[colback=redbg, colframe=black!40, boxrule=0.6pt, arc=2mm,
    left=3mm, right=3mm, top=2mm, bottom=2mm, title=\textbf{Tier R --- Remediate}]
\textbf{Context: Comparing plant growth.}
Fertilizer A plants: $\mu_1 = 24$ cm, $\sigma_1 = 4$ cm, $n_1 = 25$ (pop.\ normal).
Fertilizer B plants: $\mu_2 = 20$ cm, $\sigma_2 = 5$ cm, $n_2 = 25$ (pop.\ normal).

\begin{enumerate}
  \item Check all conditions for using the normal model for $\bar{x}_1 - \bar{x}_2$.
        \ansline{Random: stated.}\\\ansline{10\%: both $\le 10\%$ of all plants. \quad Independent groups.}\\\ansline{Normal/Large: both populations normal. All conditions met.}
  \item Find $\mu_{\bar{x}_1-\bar{x}_2}$ and $\sigma_{\bar{x}_1-\bar{x}_2}$.
        \workans{0.6in}{$\mu = 24 - 20 = 4$ cm; \quad $\sigma = \sqrt{16/25 + 25/25} = \sqrt{0.64 + 1.00} = \sqrt{1.64} \approx 1.281$ cm}
  \item Find $P(\bar{x}_1 - \bar{x}_2 > 6)$.
        \workans{0.8in}{$z = (6-4)/1.281 \approx 1.562$\\ $P(\bar{x}_1 - \bar{x}_2 > 6) \approx 1 - 0.9409 = 0.0591 \approx 5.9\%$}
\end{enumerate}
\end{tcolorbox}

\vspace{0.12in}

\begin{tcolorbox}[colback=redbg, colframe=black!40, boxrule=0.6pt, arc=2mm,
    left=3mm, right=3mm, top=2mm, bottom=2mm, title=\textbf{Tier A --- Approaching Proficiency}]
\textbf{Context: Comparing study methods.}
Students using Method A (active recall): $\mu_1 = 82$, $\sigma_1 = 10$, $n_1 = 40$.
Students using Method B (re-reading): $\mu_2 = 76$, $\sigma_2 = 12$, $n_2 = 50$.
Both populations are approximately normal.

\begin{enumerate}
  \item Check all conditions.
        \ansline{Random: stated.}\\\ansline{10\%: both $\le 10\%$ of student populations. \quad Independent groups.}\\\ansline{Normal/Large: both populations approx.\ normal; also $n_1 = 40 \ge 30$, $n_2 = 50 \ge 30$. Met.}
  \item Find $\mu$ and $\sigma$ for $\bar{x}_1 - \bar{x}_2$.
        \workans{0.5in}{$\mu = 82 - 76 = 6$; \quad $\sigma = \sqrt{100/40 + 144/50} = \sqrt{2.50 + 2.88} = \sqrt{5.38} \approx 2.320$}
  \item Find $P(\bar{x}_1 - \bar{x}_2 > 10)$.
        \workans{0.8in}{$z = (10-6)/2.320 \approx 1.724$\\ $P(\bar{x}_1 - \bar{x}_2 > 10) \approx 1 - 0.9576 = 0.0424 \approx 4.2\%$}
  \item Find $P(\bar{x}_1 - \bar{x}_2 < 0)$ and interpret. What does a negative difference
        mean in this context?
        \workans{0.9in}{$z = (0-6)/2.320 \approx -2.586$; \quad $P(\bar{x}_1 - \bar{x}_2 < 0) \approx 0.0049 \approx 0.5\%$\\ A negative difference means the re-reading group outscored the active recall group ---\\ very unlikely given the true difference, but possible due to sampling variability.}
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=redbg, colframe=black!40, boxrule=0.6pt, arc=2mm,
    left=3mm, right=3mm, top=2mm, bottom=2mm, title=\textbf{Tier E --- Extension}]
\textbf{Context: City commute times.}
City 1 commuters (right-skewed): $\mu_1 = 35$ min, $\sigma_1 = 12$, $n_1 = 60$.
City 2 commuters (right-skewed): $\mu_2 = 28$ min, $\sigma_2 = 9$, $n_2 = 45$.

\begin{enumerate}
  \item Check all conditions, including whether CLT applies to each group.
        \ansline{Random: stated.}\\\ansline{10\%: both $\le 10\%$ of city commuters. \quad Independent: separate cities.}\\\ansline{Normal/Large: $n_1 = 60 \ge 30$, $n_2 = 45 \ge 30$; CLT applies to both. All met.}
  \item Find $\mu$ and $\sigma$ for $\bar{x}_1 - \bar{x}_2$.
        \workans{0.5in}{$\mu = 35 - 28 = 7$ min; \quad $\sigma = \sqrt{144/60 + 81/45} = \sqrt{2.40 + 1.80} = \sqrt{4.20} \approx 2.049$ min}
  \item Find $P(5 < \bar{x}_1 - \bar{x}_2 < 10)$.
        \workans{0.8in}{$z_1 = (5-7)/2.049 \approx -0.976$; \quad $z_2 = (10-7)/2.049 \approx 1.464$\\ $P \approx 0.9285 - 0.1647 = 0.7638 \approx 76.4\%$}
  \item The 2.5th percentile of the sampling distribution of $\bar{x}_1 - \bar{x}_2$:
        find this value and interpret.
        \workans{0.8in}{$z = \text{invNorm}(0.025) \approx -1.960$; \quad $c = 7 + (-1.960)(2.049) \approx 2.98$ min\\ In 2.5\% of repeated samples, the difference in sample mean commutes is below 2.98 minutes.}
  \item If both $n_1$ and $n_2$ are doubled, describe what happens to $\sigma_{\bar{x}_1-\bar{x}_2}$.
        \workans{0.6in}{Each variance term is halved, so $\sigma_{\bar{x}_1-\bar{x}_2}$ is multiplied by $1/\sqrt{2} \approx 0.707$\\ --- it decreases by about 29\%.}
\end{enumerate}
\end{tcolorbox}
