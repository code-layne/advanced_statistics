\begin{enumerate}

  \item A national survey finds that 48\% of men and 61\% of women report reading for
        pleasure at least once a week ($p_1 = 0.48$ for men, $p_2 = 0.61$ for women).
        Random samples of 200 men ($n_1 = 200$) and 220 women ($n_2 = 220$) are selected.
        \begin{enumerate}[label=\alph*.]
          \item Check all conditions for using the normal model for $\hat{p}_1 - \hat{p}_2$.
                \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw1a, type=rubric, points=3, display={Both random; each sample $\le 10\%$ of its population; independent groups; counts 96, 104, 134.2, 85.8 all $\ge 10$.}]{\linewidth+6pt}{\gradedlinesheight{2}}}\writelines{2}}
          \item Find $\mu_{\hat{p}_1-\hat{p}_2}$ and $\sigma_{\hat{p}_1-\hat{p}_2}$.\gradedbelow[r]{0.5in-0.33in}{$\mu = $~\gradedblank[id=hw1b1, type=short, points=1, answer=-0.13, check=numeric, tolerance=0]{0.9in}{0.28in}\quad $\sigma = $~\gradedblank[id=hw1b2, type=short, points=1, answer=0.0483, check=numeric, tolerance=0.0005]{0.9in}{0.28in}}
                \vspace{0.5in}
          \item Find $P(\hat{p}_1 - \hat{p}_2 < -0.20)$.\gradedbelow[r]{0.9in-0.33in}{\gradedblank[id=hw1c, type=short, points=2, answer=0.0735, check=numeric, tolerance=0.003]{0.9in}{0.28in}}
                \vspace{0.9in}
          \item Interpret your answer to (c) in context.
                \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw1d, type=rubric, points=2, display={About a 7.4\% chance that the men's sample proportion is at least 20 points below the women's.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{1}}
        \end{enumerate}

  \item In one region, 25\% of households compost ($p_1 = 0.25$, $n_1 = 160$). In a
        second region, 18\% compost ($p_2 = 0.18$, $n_2 = 200$). The samples are
        independent random samples from large populations.
        \begin{enumerate}[label=\alph*.]
          \item Check the four Large Counts conditions.\gradedbelow[r]{0.04in}{\gradedblank[id=hw2a, type=rubric, points=2, display={$160(0.25) = 40$, $160(0.75) = 120$, $200(0.18) = 36$, $200(0.82) = 164$: all $\ge 10$.}]{\linewidth}{0.42in}}
                \vspace{0.5in}
          \item Find $P(\hat{p}_1 - \hat{p}_2 > 0.12)$.\gradedbelow[r]{0.9in-0.33in}{\gradedblank[id=hw2b, type=short, points=2, answer=0.1263, check=numeric, tolerance=0.003]{0.9in}{0.28in}}
                \vspace{0.9in}
        \end{enumerate}

\end{enumerate}

\begin{extensionbox}
\textbf{Extension (optional).}
Suppose $p_1 = p_2 = 0.50$ and $n_1 = n_2 = n$. Show algebraically that
$\sigma_{\hat{p}_1-\hat{p}_2} = \sqrt{1/n}$ and find the smallest $n$ such that
$\sigma_{\hat{p}_1-\hat{p}_2} \le 0.05$.
\end{extensionbox}
