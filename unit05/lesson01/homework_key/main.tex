\begin{enumerate}

  \item For each situation, identify the \textbf{statistic} and the \textbf{parameter}.
        Write the appropriate notation for each ($\bar{x}$, $\mu$, $\hat{p}$, or $p$).

        \begin{enumerate}[label=\alph*.]
          \item A researcher records the resting heart rate of a random sample of 80
                adults. The mean heart rate in the sample is 72 bpm. The researcher
                wants to estimate the mean heart rate of all adults.
                \begin{itemize}
                  \boxguard[4]
                  \item \makebox[1.8cm][l]{Statistic:} \makebox[3.2in][l]{\smash{\raisebox{-6pt}{\gradedblank[id=hw1a1, type=short, points=1, answer={sample mean 72}, check=text, display={\bfseries sample mean 72}]{3in}{0.3in}}}} \quad Notation: \makebox[1.8cm][l]{\vphantom{\blank{1.8cm}}\hspace{46bp}\smash{\gradedblank[id=hw1a2, type=short, points=1, answer=\bar{x}, check=expression, display={\boldmath$\bar{x}$}]{0.5in}{0.19in}}}\rule[-9pt]{0pt}{22pt}
                  \boxguard[4]
                  \item \makebox[1.8cm][l]{Parameter:} \makebox[3.2in][l]{\smash{\raisebox{-6pt}{\gradedblank[id=hw1a3, type=short, points=1, answer={mean heart rate of all adults}, check=text, display={\bfseries mean of all}]{3in}{0.3in}}}} \quad Notation: \makebox[1.8cm][l]{\vphantom{\blank{1.8cm}}\hspace{46bp}\smash{\gradedblank[id=hw1a4, type=short, points=1, answer=\mu, check=expression, display={\boldmath$\mu$}]{0.5in}{0.19in}}}\rule[-9pt]{0pt}{22pt}
                \end{itemize}
          \item In a survey of 200 randomly selected college students, 64 said they
                exercise at least 3 times per week. The survey is meant to estimate the
                proportion of all college students who do so.
                \begin{itemize}
                  \boxguard[4]
                  \item \makebox[1.8cm][l]{Statistic:} \makebox[3.2in][l]{\smash{\raisebox{-6pt}{\gradedblank[id=hw1b1, type=short, points=1, answer={sample proportion 0.32}, check=text, display={\bfseries sample prop. 0.32}]{3in}{0.3in}}}} \quad Notation: \makebox[1.8cm][l]{\vphantom{\blank{1.8cm}}\hspace{46bp}\smash{\gradedblank[id=hw1b2, type=short, points=1, answer=\hat{p}, check=expression, display={\boldmath$\hat{p}$}]{0.5in}{0.19in}}}\rule[-9pt]{0pt}{22pt}
                  \boxguard[4]
                  \item \makebox[1.8cm][l]{Parameter:} \makebox[3.2in][l]{\smash{\raisebox{-6pt}{\gradedblank[id=hw1b3, type=short, points=1, answer={proportion of all college students who exercise 3+ times a week}, check=text, display={\bfseries prop. of all}]{3in}{0.3in}}}} \quad Notation: \makebox[1.8cm][l]{\vphantom{\blank{1.8cm}}\hspace{46bp}\smash{\gradedblank[id=hw1b4, type=short, points=1, answer=p, check=expression, display={\boldmath$p$}]{0.5in}{0.19in}}}\rule[-9pt]{0pt}{22pt}
                \end{itemize}
        \end{enumerate}

  \boxguard[22]
  \item A botanist measures the heights of 15 randomly selected seedlings from a
        greenhouse and finds a mean height of 4.3 cm. Another researcher measures 15
        different randomly selected seedlings from the same greenhouse and gets 4.7 cm.

        \begin{enumerate}[label=\alph*.]
          \item Are these two measurements the same thing? Explain. \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2a, type=rubric, points=2, display={No: two different samples of 15 seedlings; each mean describes only its own sample.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}
          \boxguard[17]
          \item Is either measurement necessarily wrong? Why or why not? \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2b, type=rubric, points=2, display={No: different random samples give different means by chance (sampling variability).}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}
          \item What do you call the variation between these two measurements?
                \makebox[5cm][l]{\vphantom{\blank{5cm}}\hspace{46bp}\smash{\gradedblank[id=hw2c, type=short, points=1, answer={sampling variability}, check=text]{1.5in}{0.17in}}}
        \end{enumerate}

  \boxguard[8]
  \item A simulation takes 500 random samples of size $n = 40$ from a population with
        mean $\mu = 100$ and records each sample mean. The 500 sample means are then
        displayed in a dot-plot.

        \begin{enumerate}[label=\alph*.]
          \item What is the name of the distribution shown in the dot-plot?
                \makebox[7cm][l]{\vphantom{\blank{7cm}}\hspace{46bp}\smash{\gradedblank[id=hw3a, type=short, points=1, answer={sampling distribution of the sample mean}, check=text]{2.1in}{0.17in}}}
          \boxguard[17]
          \item Would you expect the center of the dot-plot to be near 100? Explain.
                \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw3b, type=rubric, points=2, display={Yes: sample means vary around $\mu = 100$; the sampling distribution's center equals the population mean.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}
          \boxguard[17]
          \item If the sample size were increased to $n = 200$, how would you expect
                the spread of the dot-plot to change? \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw3c, type=rubric, points=2, display={The spread would decrease: larger samples mean less sampling variability.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}
        \end{enumerate}

\end{enumerate}

\begin{extensionbox}
\textbf{Extension (optional).} Find a news article that reports two different survey
results about the same topic (e.g., two polls about political approval ratings, consumer
confidence, or vaccine attitudes). Write a short paragraph (3--4 sentences) explaining
whether the difference between the two results could be due to sampling variability or
might indicate a real difference in the populations surveyed.
\end{extensionbox}
