\begin{notesbox}{Part B --- What $r$ Can and Cannot Tell You}
\interlinepenalty=10000 % keep each prompt whole when the box breaks
\begin{enumerate}[leftmargin=1.9em, itemsep=2pt, topsep=2pt]
  \item \textbf{AP-style multiple choice.} Which correlation shows the \emph{strongest}
        linear association? Write its letter in the Letter box, then justify it in a sentence.\hfill\gradedright{\textit{Letter:}~\gradedblank[id=hw3, type=choice, points=1, answer=B]{0.45in}{0.24in}}\rule[-0.2in]{0pt}{0.48in}\par
        \textbf{(A)} $r = 0.45$ \quad \textbf{(B)} $r = -0.91$ \quad \textbf{(C)} $r = 0$
        \quad \textbf{(D)} $r = 0.78$ \quad \textbf{(E)} $r = -0.20$
        \answerspace{2.1in}{\normalcolor\mdseries\hspace*{12.5pt}\gradedblank[id=hw3j, type=rubric, points=1, display={$-0.91$ is closest to $\pm 1$; the sign gives only the direction.}]{\linewidth-12.5pt}{0.6cm}}
  \item For 40 people aged 5 to 85, age vs.\ reaction time makes a clear U (young adults
        fastest), and $r = 0.05$. Jordan says age and reaction time are unrelated. Is he right?
        \answerspace{2.1in}{\normalcolor\mdseries\hspace*{12.5pt}\gradedblank[id=hw4, type=rubric, points=2, display={No. There is a strong \emph{curved} relationship. $r$ near 0 only
        says there is no \emph{linear} association.}]{\linewidth-12.5pt}{0.9cm}}
  \item Across 50 towns, $r = 0.92$ between the number of libraries and the number of crimes.
        Should a town close libraries to cut crime? Explain.
        \answerspace{2.1in}{\normalcolor\mdseries\hspace*{12.5pt}\gradedblank[id=hw5, type=rubric, points=2, display={No. Correlation is not causation. Bigger towns have more of both, so
        population is a likely lurking variable.}]{\linewidth-12.5pt}{0.9cm}}
\end{enumerate}
\end{notesbox}
