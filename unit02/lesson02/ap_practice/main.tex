\textbf{Section I --- Multiple Choice.} Circle the single best answer.
\end{headlinebox}

\begin{enumerate}[leftmargin=1.9em, itemsep=0.06in, topsep=4pt]

  \item Which is the best interpretation of $r = -0.72$?
        \begin{enumerate}[label=\textbf{(\Alph*)}, leftmargin=2.2em, itemsep=0pt, topsep=2pt]
          \item 72\% of the change in sleep is caused by screen time.
          \item There is a moderately strong, negative, linear association.
          \item Each extra hour of screen time lowers sleep by 0.72 hours.
          \item The association is weak, because $r$ is negative.
          \item Screen time and sleep are unrelated, because $r$ is not close to $-1$.
        \end{enumerate}

  \item Which change would alter the value of $r$?
        \begin{enumerate}[label=\textbf{(\Alph*)}, leftmargin=2.2em, itemsep=0pt, topsep=2pt]
          \item Recording sleep in minutes instead of hours.
          \item Putting sleep on the $x$-axis and screen time on the $y$-axis.
          \item Adding 30 minutes to every teen's screen time.
          \item Adding a 21st teen with 9 hours of screen time and 9 hours of sleep.
          \item Giving every teen a new ID number.
        \end{enumerate}

\end{enumerate}
