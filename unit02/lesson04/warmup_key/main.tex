\begin{enumerate}[leftmargin=1.9em, itemsep=0.70in]

  \item \textbf{From last lesson.} A pizza shop models delivery time with
        $\widehat{\text{minutes}} = 8 + 1.5(\text{miles})$.

        \vspace{12pt}
        Predicted time for a 4-mile delivery: \ans{14 minutes}
        \vspace{16pt}

        That delivery actually took 16 minutes. Its residual: \ans{$16 - 14 = 2$ min}

  \item Two lines were fit to the \emph{same} four points. Here are their residuals.

        \vspace{12pt}
        \renewcommand{\arraystretch}{1.5}
        \begin{tabularx}{\linewidth}{@{} l Y Y Y Y | Y Y @{}}
        \rowcolor{sky}
        & \multicolumn{4}{c|}{\textbf{Residuals}} & \textbf{Sum} & \textbf{Sum of squares} \\
        \hline
        \textbf{Line A} & 3 & $-3$ & 1 & $-1$ & \ans{0} & \ans{20} \\
        \textbf{Line B} & 2 & $-2$ & 2 & $-2$ & \ans{0} & \ans{16} \\
        \end{tabularx}

        \vspace{14pt}
        Which line fits better? \ans{Line B} \quad Why can't the plain \textbf{Sum} column
        tell you?\\[18pt]
        \ansline{Both sums are 0 --- the positive and negative residuals cancel out.}

  \item For 20 students, the correlation between hours of sleep and reaction time is
        $r = -0.80$.

        \vspace{12pt}
        Describe the direction and strength: \ans{negative, fairly strong}
        \vspace{16pt}

        Compute: \quad $(-0.80)^2 =$ \ans{0.64} \qquad
        $-0.80 \times \dfrac{45}{1.5} =$ \ans{$-24$}

\end{enumerate}
