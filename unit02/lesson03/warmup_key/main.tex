\begin{enumerate}[leftmargin=1.9em, itemsep=1.0in]

  \item \textbf{From last lesson.} A pizza shop plotted delivery distance (miles) against
        delivery time (minutes) for 17 deliveries. The points rise from left to right in a
        tight, straight band.

        \vspace{12pt}
        Explanatory variable: \ans{distance} \qquad Direction: \ans{positive}

        \vspace{16pt}
        Is $r$ closer to $0.3$ or to $0.95$? \ans{$0.95$} \qquad Form: \ans{linear}

  \item Use the line $y = 8 + 3x$.

        \vspace{12pt}
        When $x = 4$, \quad $y =$ \ans{$8+3(4)=20$} \qquad When $x = 7$, \quad $y =$ \ans{$8+3(7)=29$}

  \item The point $(5,\ 26)$ is plotted beside the line $y = 8 + 3x$.

        \vspace{12pt}
        At $x = 5$ the line is at height \ans{$8+3(5)=23$}

        \vspace{16pt}
        Is the point above or below the line? \ans{above} \quad By how much? \ans{3}

        \vspace{16pt}
        Now the point $(4,\ 17)$: above or below? \ans{below} \quad By how much? \ans{3}

        \vspace{16pt}
        Which point's actual $y$ came out \emph{bigger} than the line predicted?\\[18pt]
        \ansline{$(5,\ 26)$ --- its $y$ is 3 above the line; $(4,\ 17)$ is 3 below it.}

\end{enumerate}
