% ── ROW 1: THE EQUATION & PREDICTION ─────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{The regression line predicts}
  \sectionlabel{notes, The Equation \& Prediction $\cdot$ problems 1--3}
  \vspace{2pt}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{Regression line}
        $\hat{y} = a + bx$ uses the explanatory variable $x$ to predict the response $y$
        (EK DAT-1.D.1--2).
      \end{block}
      \vspace{4pt}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $\hat{y}$ --- \textbf{predicted}, from the line.
        \item $y$ --- \textbf{actual}, from the data.
        \item Pizza shop: $\widehat{\text{time}} = 8 + 3(\text{distance})$.\\

% ── ROW 2: EXTRAPOLATION ─────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Predicting beyond the data}
  \sectionlabel{notes, Predicting Beyond the Data $\cdot$ problems 4--6}
  \vspace{10pt}
  \begin{block}{Extrapolation (EK DAT-1.D.3)}
    Predicting for an $x$-value \textbf{outside} the interval of $x$-values used to build the
    line. The farther out we go, the \textbf{less} reliable the prediction.
  \end{block}
  \vspace{8pt}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \textbf{\color{navy}Inside 1--9 miles}\\[3pt]
      3 miles, 8.5 miles --- the line has data on both sides to check against.
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{\color{navy}Outside 1--9 miles}\\[3pt]
      30 miles: $8 + 3(30) = 98$ minutes. Highways? A second driver? We have
      \redink{no deliveries out there} to know.
    \end{column}
  \end{columns}
  \vspace{12pt}
  \begin{center}
    {\large The equation will always give you a number. \redink{Say whether to trust it.}}
  \end{center}
\end{frame}

% ── ROW 3: RESIDUALS — THE CRUX ──────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Actual minus predicted}
  \sectionlabel{notes, Actual Minus Predicted $\cdot$ problems 7--10 $\cdot$ the crux is 9}
  \vspace{2pt}
  \begin{columns}[t]
    \begin{column}{0.54\textwidth}
      \begin{block}{Residual (EK DAT-1.E.1)}
        \centering residual $= y - \hat{y}$ \quad = \quad actual $-$ predicted
      \end{block}
      \vspace{4pt}
      {\small The 5-mile delivery took 26 minutes: $26 - 23 =$ \redink{+3}.\\[3pt]
      \textbf{Positive} $\to$ took \emph{longer} than predicted $\to$ above the line $\to$ the
      line predicted too \redink{low}.\\[3pt]
      \textbf{Negative} $\to$ below the line $\to$ the line predicted too high.}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{center}
      {\scriptsize\bfseries Residual plot}\\[2pt]
      \begin{tikzpicture}[xscale=0.42, yscale=0.32]
        \draw[->, thick] (0,-4.6) -- (0,4.8);
        \draw[thick] (0,-4.6) -- (10.2,-4.6);
        \draw[dashed, slate] (0,0) -- (10.2,0);
        \foreach \y in {-4,-2,0,2,4} {\draw (-0.12,\y) -- (0.12,\y);
          \node[left, font=\tiny] at (0,\y) {\y};}
        \foreach \x in {0,2,4,6,8,10} {\draw (\x,-4.75) -- (\x,-4.45);
          \node[below, font=\tiny] at (\x,-4.6) {\x};}
        \foreach \x/\e in {1/1, 1.5/-2, 2/2, 2.5/-1, 3/0, 3.5/2, 4/-3, 4.5/1, 5/3, 5.5/-2,
                           6/-2, 6.5/1, 7/-1, 7.5/2, 8/0, 8.5/-2, 9/1}
          {\fill[navy] (\x,\e) circle[x radius=0.13, y radius=0.17];}
        \node[font=\tiny] at (5,-6.3) {Distance (miles)};
      \end{tikzpicture}\\[2pt]
      {\scriptsize Random scatter around 0, no pattern $\to$ a line is appropriate
      (EK DAT-1.F.1)}
      \end{center}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \begin{center}
    {\small Jordan: ``$23 - 26 = -3$, so the line predicted too high.'' \quad
    \redink{Two mistakes. Find both.}}
  \end{center}
\end{frame}

% ── WE DO ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Guided Practice: Sleep \& Reaction}
  \sectionlabel[goldacc]{We do $\cdot$ you hold the pen $\cdot$ problems 11--14}
  Twenty adults who slept 4 to 9 hours: \quad
  $\widehat{\text{RT}} = 430 - 25(\text{sleep})$, reaction time in milliseconds.
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item Predict the reaction time after 7 hours of sleep.
    \item One adult slept 7 hours and reacted in 270 ms. Residual? Too high or too low?
    \item Predict for 12 hours of sleep. Would you trust it?
    \item Does the residual plot say a line is appropriate?
  \end{itemize}
  \begin{block}{The question behind the question}
    A reaction time \emph{above} the prediction is a \emph{slower} reaction. Say what the
    sign means in \redink{this} context before you say ``good'' or ``bad.''
  \end{block}
\end{frame}

% ── DEBRIEF ──────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Debrief: which way does the residual point?}
  \sectionlabel{8 minutes $\cdot$ pens down, papers up --- we say these aloud}
  \vspace{6pt}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{The order is the definition}
        Actual minus predicted. $\hat{y} - y$ flips every sign and every conclusion.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Positive means under}
        Point above the line $\Rightarrow$ the line guessed too \redink{low}.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{12pt}
  \begin{block}{Two things to say out loud}
    Why is a 12-hour prediction a guess even though the equation gives a number? And what
    would a residual plot have to look like for us to \emph{stop} trusting the line?
  \end{block}
\end{frame}
