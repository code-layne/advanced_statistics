\item \textbf{Sleep and Grade Level.}

\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{@{} l c c c @{}}
\hline
           & \textbf{$<$7 hrs} & \textbf{7$+$ hrs} & \textbf{Total} \\
\hline
9th grade  & 30 & 30 & 60  \\
10th grade & 38 & 22 & 60  \\
11th grade & 42 & 18 & 60  \\
\hline
\textbf{Total} & 110 & 70 & 180 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}[label=\alph*., itemsep=6pt]
  \item \textbf{State.}
        Test type: \ans{Independence --- one random sample of 180 teens, classified by two variables (grade and sleep amount).}\gradedkey{\gradedblank[id=hw1a1, type=short, points=1, answer={independence}, check=text, display={\bfseries Independence}]{3in}{0.3in}}

        $H_0$: \ansline{There is no association between grade level and sleep amount for teens at the school.}\gradedkey{\gradedblank[id=hw1a2, type=short, points=1, answer={no association}, check=text, display={\bfseries No association.}]{2.3in}{0.3in}}

        $H_a$: \ansline{There is an association between grade level and sleep amount for teens at the school.}\gradedkey{\gradedblank[id=hw1a3, type=short, points=1, answer={association}, check=text, display={\bfseries Association.}]{2.3in}{0.3in}}

  \boxguard[16]
  \item \textbf{Plan.} Method: \ans{Chi-square test for independence.}\gradedkey{\gradedblank[id=hw1b1, type=short, points=1, answer={chi-square test for independence}, check=text, display={\bfseries $\chi^2$ test for independence}]{3.5in}{0.3in}}

        \boxguard[14]
        Random: \ansline{Random selection stated in the problem.}\gradedkeyline{\gradedblank[id=hw1b2, type=rubric, points=1, display={\bfseries Stated.}]{\linewidth+6pt}{\gradedlinesheight{9}}}

        $E_{\text{9th, }<7} = \dfrac{60 \times 110}{180} = $ \ans{36.67}
        \quad $E_{\text{9th, }7+} = \dfrac{60 \times 70}{180} = $ \ans{23.33}\gradedkey{$E_{\text{9th, }<7}$~\gradedblank[id=hw1b3, type=short, points=1, answer=36.67, check=numeric, tolerance=0.01]{0.95cm}{0.2in}\quad $E_{\text{9th, }7+}$~\gradedblank[id=hw1b4, type=short, points=1, answer=23.33, check=numeric, tolerance=0.01]{1.6cm}{0.2in}}

        \ans{All three grade levels have the same expected counts (36.67 and 23.33), all $\ge 5$. Condition met.}\gradedkey{\gradedblank[id=hw1b5, type=short, points=1, answer=yes, check=text]{1.2cm}{0.2in}}

  \newpage
  \item \textbf{Do.} $\text{df} = ($ \ans{3} $-1)($ \ans{2} $-1) = $ \ans{2}\gradedkey{\gradedblank[id=hw1c1, type=short, points=1, answer=2, check=numeric, tolerance=0]{1.2cm}{0.2in}}

        \renewcommand{\arraystretch}{1.6}
        \begin{center}
        \begin{tabular}{@{} l c c @{}}
        \hline
                   & \textbf{$<$7 hrs} & \textbf{7$+$ hrs} \\
        \hline
        9th grade  & $\dfrac{(30-{\color{keyred}\mathbf{36.67}})^2}{{\color{keyred}\mathbf{36.67}}} \approx 1.21$
                   & $\dfrac{(30-{\color{keyred}\mathbf{23.33}})^2}{{\color{keyred}\mathbf{23.33}}} \approx 1.91$ \\
        10th grade & \ans{$\dfrac{(38-36.67)^2}{36.67} \approx 0.05$} & \ans{$\dfrac{(22-23.33)^2}{23.33} \approx 0.08$} \\
        11th grade & \ans{$\dfrac{(42-36.67)^2}{36.67} \approx 0.78$} & \ans{$\dfrac{(18-23.33)^2}{23.33} \approx 1.22$} \\
        \hline
        \end{tabular}
        \end{center}

        $\chi^2 = 1.21 + 1.91 + 0.05 + 0.08 + 0.78 + 1.22 \approx $ \ans{5.25}\gradedkey{\gradedblank[id=hw1c2, type=short, points=2, answer=5.25, check=numeric, tolerance=0.1]{2.2cm}{0.2in}}

        \begin{teachernote}
        Exact: $(30-36.\overline{6})^2/36.\overline{6} + \cdots$. With exact fractions $\chi^2 = 5.25$; rounding intermediate steps may give 5.25--5.33. Using the table: df = 2, $\chi^2 = 5.25$ falls between 4.605 ($p = 0.10$) and 5.991 ($p = 0.05$), so the p-value is between 0.05 and 0.10. Accept $p \approx 0.07$ from technology.
        \end{teachernote}

        p-value: \ans{Between 0.05 and 0.10 (from table, df = 2: $\chi^2 = 5.25$ lies between 4.605 and 5.991); technology gives $p \approx 0.072$.}\gradedkey{\gradedblank[id=hw1c3, type=short, points=1, answer={0.05 to 0.10}, check=text, display={\bfseries 0.05 to 0.10}]{1.8in}{0.3in}}

  \item \textbf{Conclude.} At $\alpha = 0.05$:
        \ansline{Because $p \approx 0.07 > \alpha = 0.05$, we fail to reject $H_0$. We do not have convincing evidence of an association between grade level and sleep amount.}\gradedkeyline{\gradedblank[id=hw1d, type=rubric, points=2, display={$p \approx 0.07 > 0.05$: fail to reject $H_0$; no convincing evidence of an association between grade and sleep.}]{\linewidth+6pt}{\gradedlinesheight{1}}}
        \ansline{}
\end{enumerate}

\item \textbf{Conceptual Question.}

\ansline{A larger $\chi^2$ value means the observed counts deviate more from the expected counts we would predict if $H_0$ were true. These larger deviations are less likely to occur by random chance alone. In terms of the chi-square distribution, a larger test statistic falls farther in the right tail, resulting in a smaller p-value. A smaller p-value means the data provide stronger evidence against $H_0$, making it more likely that the variables are truly associated (or the distributions differ).}\gradedkeyline{\gradedblank[id=hw3, type=rubric, points=3, display={Larger $\chi^2$ = observed counts farther from expected under $H_0$; farther into the right tail, so a smaller p-value and stronger evidence against $H_0$.}]{\linewidth+6pt}{\gradedlinesheight{3}}}
\ansline{}
