\textbf{Problem 2.} For each scenario below, state the appropriate H$_0$ and H$_a$
in terms of $\beta$, and explain what $\beta = 0$ means in context.

\begin{enumerate}[label=\alph*.]

  \item $x = $ daily rainfall (inches), $y = $ umbrella sales, $n = 60$ days
        (random sample).

        H$_0$: \makebox[7cm][l]{\hspace{1.7cm}\smash{\gradedblank[id=hw2a1, type=rubric, points=1, display={\bfseries\boldmath $\beta = 0$}]{4cm}{0.2in}}} \quad H$_a$: \makebox[7cm][l]{\hspace{1.7cm}\smash{\gradedblank[id=hw2a2, type=rubric, points=1, display={\bfseries\boldmath $\beta \ne 0$}]{4cm}{0.2in}}}

        $\beta = 0$ means: \gradedhide{\writelines{1}\gradedright{\smash{\gradedblank[id=hw2a3, type=rubric, points=1, display={daily rainfall has no linear effect on umbrella sales in the population.}]{5in}{0.2in}}}}

  \item $x = $ age (years), $y = $ marathon finish time (minutes), $n = 45$ runners
        (random sample). The researcher suspects older runners take longer, so she
        considers a one-sided test.

        H$_0$: \makebox[7cm][l]{\hspace{1.7cm}\smash{\gradedblank[id=hw2b1, type=rubric, points=1, display={\bfseries\boldmath $\beta = 0$}]{4cm}{0.2in}}} \quad H$_a$: \makebox[7cm][l]{\hspace{1.7cm}\smash{\gradedblank[id=hw2b2, type=rubric, points=1, display={\bfseries\boldmath $\beta > 0$}]{4cm}{0.2in}}}

        $\beta = 0$ means: \gradedhide{\writelines{1}\gradedright{\smash{\gradedblank[id=hw2b3, type=rubric, points=1, display={age has no linear effect on marathon finish time in the population.}]{5in}{0.2in}}}}

  \item $x = $ class size, $y = $ average exam score, $n = 35$ classes (random sample).
        A teacher claims larger classes score lower ($\beta < -1$). State hypotheses
        to test this claim.

        H$_0$: \makebox[7cm][l]{\hspace{1.7cm}\smash{\gradedblank[id=hw2c1, type=rubric, points=1, display={\bfseries\boldmath $\beta = 0$}]{4cm}{0.2in}}} \quad H$_a$: \makebox[7cm][l]{\hspace{1.7cm}\smash{\gradedblank[id=hw2c2, type=rubric, points=1, display={\bfseries\boldmath $\beta < 0$}]{4cm}{0.2in}}}

        $\beta = 0$ means: \gradedhide{\writelines{1}\gradedright{\smash{\gradedblank[id=hw2c3, type=rubric, points=1, display={class size has no linear effect on average exam score in the population.}]{5in}{0.2in}}}}

\end{enumerate}
