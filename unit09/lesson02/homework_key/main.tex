\textbf{Problem 1: Sleep and Reaction Time}

\vspace{0.05in}
A random sample of 22 college students was used to study the relationship between hours
of sleep per night ($x$) and reaction time in milliseconds ($y$). Computer output is
shown below.

\vspace{0.1in}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lrrrr}
    \textbf{Predictor} & \textbf{Coef} & \textbf{SE Coef} & \textbf{T} & \textbf{P} \\
    \hline
    Constant & 412.3 & 28.4 & 14.52 & 0.000 \\
    Sleep    & $-18.6$ & 5.3 & $-3.51$ & 0.002 \\
    \hline
    \multicolumn{5}{l}{$s = 24.1$ \quad $n = 22$}
\end{tabular}
\end{center}

\vspace{0.1in}
\begin{enumerate}[label=\alph*.]

    \item \ans{$b = -18.6$ ms per hour of sleep;\quad $SE_b = 5.3$.}\gradedkey{$b = $~\gradedblank[id=hw1a1, type=short, points=1, answer=-18.6, check=numeric, tolerance=0]{0.75in}{0.2in}\quad $SE_b = $~\gradedblank[id=hw1a2, type=short, points=1, answer=5.3, check=numeric, tolerance=0]{0.75in}{0.2in}}

    \item \gradedtop{\gradedblank[id=hw1b, type=rubric, points=3, display={\textbf{L}: Random scatter in residual plot --- linear condition met. \textbf{I}: Random sample of college students; $n = 22 \le 10\%$ of all students --- independent. \textbf{N}: $n = 22$; problem states conditions are met --- normal condition assumed. \textbf{E}: Roughly constant spread in residual plot --- equal variance met. \textbf{R}: Random sample --- random condition met.}]{\linewidth+6pt}{\gradedlinesheight{4}}}

    \item \ans{$CI: -18.6 \pm 2.086(5.3) = -18.6 \pm 11.056 = (-29.656,\ -7.544)$ ms per hour.
          Round: $(-29.7,\ -7.5)$ ms per hour of sleep.}\gradedkey{(~\gradedblank[id=hw1c1, type=short, points=1, answer=-29.7, check=numeric, tolerance=0.1]{0.8in}{0.2in},~\gradedblank[id=hw1c2, type=short, points=1, answer=-7.5, check=numeric, tolerance=0.1]{0.8in}{0.2in}~)}

    \item \ans{We are 95\% confident that the true slope of the population regression line
          relating reaction time to hours of sleep for college students is between $-29.7$
          and $-7.5$ milliseconds per hour of sleep.}\gradedkeyline{\gradedblank[id=hw1d, type=rubric, points=2, display={95\% confident the true slope (reaction time vs.\ hours of sleep, college students) is between $-29.7$ and $-7.5$ ms per hour.}]{\linewidth+6pt}{\gradedlinesheight{1}}}

    \item \gradedtop{\gradedblank[id=hw1e, type=rubric, points=2, display={\bfseries Yes: the whole CI is below 0 (0 not in it), so $\beta < 0$: more sleep, faster reactions.}]{\linewidth+6pt}{\gradedlinesheight{1}}}

\end{enumerate}

\textbf{Problem 2: Temperature and Heating Fuel Cost}

\vspace{0.05in}
A researcher studies 18 cities, recording average daily high temperature in January
($x$, ${}^\circ$F) and monthly heating fuel cost ($y$, \$/month).
Output gives $b = -4.8$, $SE_b = 1.1$, and $n = 18$.
Use $t^* = 2.120$ ($df = 16$). Assume all LINER conditions are met.

\vspace{0.1in}
\begin{enumerate}[label=\alph*.]

    \item \ans{$CI: -4.8 \pm 2.120(1.1) = -4.8 \pm 2.332 = (-7.132,\ -2.468)$ dollars per degree Fahrenheit.}\gradedkey{(~\gradedblank[id=hw2a1, type=short, points=1, answer=-7.13, check=numeric, tolerance=0.02]{0.8in}{0.2in},~\gradedblank[id=hw2a2, type=short, points=1, answer=-2.47, check=numeric, tolerance=0.02]{0.8in}{0.2in}~)}\gradedkeyline{\gradedblank[id=hw2a3, type=rubric, points=2, display={We are 95\% confident that the true slope of the population regression line relating monthly heating fuel cost to January temperature for all cities is between $-\$7.13$ and $-\$2.47$ per ${}^\circ$F.}]{\linewidth+6pt}{\gradedlinesheight{2}}}

    \item \gradedtop{\gradedblank[id=hw2b, type=rubric, points=2, display={Yes. The value $\beta = -5$ falls inside the CI $(-7.13,\ -2.47)$, so the utility company's claim is consistent with the data.}]{\linewidth+6pt}{\gradedlinesheight{1}}}

    \item \gradedtop{\gradedblank[id=hw2c, type=rubric, points=2, display={\bfseries Narrower: $SE_b = s/(s_x\sqrt{n-1})$ is smaller for larger $n$, so a smaller margin of error.}]{\linewidth+6pt}{\gradedlinesheight{1}}}

\end{enumerate}
