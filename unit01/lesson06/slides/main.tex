% ── FIVE-NUMBER SUMMARY AND THE BOXPLOT ─────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Five numbers, drawn to scale}
  \sectionlabel{five-number summary $\cdot$ boxplot}
  \vspace{2pt}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.9]
        \bxp{0.9}{0.42}{0}{0.9}{1.9}{4.1}{8.0}
        \draw[->, thick] (-0.2,0) -- (8.6,0);
        \foreach \x/\lab in {0/min, 0.9/$Q_1$, 1.9/med, 4.1/$Q_3$, 8.0/max} {
          \draw[linegray, very thin] (\x,0.05) -- (\x,1.32);
          \draw (\x,-0.08) -- (\x,0.08);
          \node[below, font=\tiny] at (\x,-0.1) {\lab};}
      \end{tikzpicture}
      \end{center}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item The \textbf{five-number summary}: minimum, $Q_1$, median, $Q_3$, maximum
              (EK UNC-1.L.1).
        \item The \textbf{box} runs $Q_1$ to $Q_3$ --- the \textbf{middle 50\%}. The line inside
              is the \textbf{median}.
        \item The \textbf{whiskers} reach the most extreme value that is \redink{not} an
              outlier (EK UNC-1.L.2).
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{6pt}
  \begin{center}
    {\small\color{slate} A boxplot is nothing but those five numbers placed on a number line.}
  \end{center}
\end{frame}

% ── OUTLIERS AND THE MODIFIED BOXPLOT ───────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{The rider who was set apart}
  \sectionlabel{modified boxplot}
  \vspace{2pt}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.9]
        \bxp{0.9}{0.42}{0.2}{0.9}{1.9}{3.4}{5.0}
        \fill[navy] (7.6,0.9) circle (0.10);
        \draw[->, thick] (-0.2,0) -- (8.6,0);
        \node[font=\tiny, above] at (7.6,1.08) {58 min};
        \node[font=\tiny, above] at (5.0,1.08) {40 min};
      \end{tikzpicture}
      \end{center}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Route B: $Q_1 = 20$, $Q_3 = 32$, so IQR $= 12$.
        \item Cut-off: $32 + 1.5(12) = \mathbf{50}$. And $58 > 50$.
        \item So 58 gets \textbf{its own symbol}, and the whisker stops at \textbf{40} --- the
              most extreme value that is not an outlier.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{6pt}
  \begin{center}
    {\small\color{slate} That is a \textbf{modified boxplot}. It is the version the AP exam
    draws.}
  \end{center}
\end{frame}

% ── COMPARING GROUPS ─────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Comparing two groups}
  \sectionlabel{parallel (side-by-side) boxplots}
  \vspace{4pt}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{The checklist (EK UNC-1.N.1, 1.O.1)}
        \textbf{Center} $\cdot$ \textbf{Variability} $\cdot$ \textbf{Unusual features} --- all
        three, every time.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{And two non-negotiables}
        A \textbf{comparative word} --- \emph{higher than}, \emph{more variable than} --- and
        the \textbf{context}: name the variable, its units, and both groups.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{12pt}
  \begin{center}
    {\large Describing group 1, then describing group 2, is \redink{not} a comparison ---
    and earns nothing.}
  \end{center}
\end{frame}

% ── MEAN VS MEDIAN, AND WHAT THE BOX HIDES ──────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Two footnotes worth keeping}
  \sectionlabel{skew and the boxplot's blind spot}
  \vspace{4pt}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Mean vs.\ median (EK UNC-1.M.2)}
        Symmetric: the two sit close together.\\[3pt]
        Skewed \textbf{right}: the mean is to the \textbf{right} of the median.\\[3pt]
        Skewed \textbf{left}: to the \textbf{left}.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{What a boxplot cannot show}
        \textbf{Gaps}, \textbf{clusters}, and a \textbf{second peak} all vanish inside the box.
        If somebody asks whether there are two kinds of night, ask for a \redink{dotplot or
        histogram} instead.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \begin{center}
    {\small\color{slate} Five numbers buy you a clean comparison. What they cost you is
    everything between them.}
  \end{center}
\end{frame}

% ── THE CRUX ─────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Why the wide section is not the full one}
  \sectionlabel{the whole point of today}
  \vspace{6pt}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Twenty players. Four sections. Your counts: \textbf{5 \quad 5 \quad 5 \quad 5}.
    \item That is \textbf{25\% \quad 25\% \quad 25\% \quad 25\%} --- and it is always true, in
          every boxplot ever drawn.
    \item The right section was more than five times as wide. It held the \redink{same five
          players}, just spread from 52 to 74 instead of packed into 30 to 33.
  \end{itemize}
  \vspace{10pt}
  \begin{block}{Essential Knowledge (UNC-1.L.2)}
    \textbf{Each section of a boxplot holds one quarter of the data.} Width reports how far
    apart those values are --- \textbf{spread}, never \textbf{count}.
  \end{block}
\end{frame}

% ── APPLICATION ──────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Application: Two Science Sections}
  \sectionlabel[goldacc]{We work this one together --- you hold the pen}
  Two sections of the same course, one test, two boxplots on one axis.
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item Read both medians and both middle-half widths.
    \item Which section is skewed, and which way? How does the picture tell you?
    \item Write the comparison as \textbf{one paragraph}, in context.
  \end{itemize}
  \begin{block}{The standard, out loud}
    A paragraph that describes Period 2 and then describes Period 5 earns \redink{nothing}.
    The words that earn the point are \emph{higher than} and \emph{more variable than} --- with
    the variable and both sections named.
  \end{block}
\end{frame}
