\begin{enumerate}[leftmargin=1.7em, itemsep=0.20in]

  \item \textbf{From last lesson.} A pet clinic records, for each animal it treats, the
        \textbf{species} (dog, cat, bird) and the animal's \textbf{weight in kilograms}.

        \vspace{5pt}
        Species is \ans{categorical} \qquad Weight is \ans{quantitative}
        \vspace{4pt}

        (Write \emph{categorical} or \emph{quantitative} in each blank.)

  \item Work these out. Write each answer as a percent.
        \vspace{6pt}

        \hspace{1.0em} 80 out of 400 is \ans{20\%}
        \hfill
        150 out of 1{,}000 is \ans{15\%}
        \hspace{1.0em}
        \vspace{8pt}

        Which of those two is the \textbf{larger share}? \ans{80 out of 400}

  \item Team A won 9 of its 12 games. Team B won 15 of its 25 games.

        \vspace{5pt}
        Which team has the better record? \ans{Team A (75\% vs.\ 60\%)}
        \vspace{6pt}

        Team B won more games than Team A. Explain why that is not what decided your
        answer.\\[4pt]
        \ansline{The teams played different numbers of games, so the raw win totals are not}\\[1pt]\ansline{comparable. What matters is the share of games won, not how many.}

\end{enumerate}
