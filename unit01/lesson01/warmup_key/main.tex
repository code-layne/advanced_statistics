\begin{hookbox}
A music teacher keeps this sheet on every student who rents an instrument.

\vspace{10pt}
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{@{} l l c c @{}}
\rowcolor{sky}
\textbf{Student} & \textbf{Instrument} & \textbf{Weeks rented} & \textbf{Locker \#} \\
\hline
Ibarra   & Clarinet & 14 & 208 \\
Okafor   & Trumpet  &  6 & 145 \\
Petrov   & Clarinet & 22 & 331 \\
Nakamura & Cello    &  9 & 117 \\
\hline
\end{tabular}
\end{center}
\vspace{4pt}
\end{hookbox}

\vspace{0.30in}

\begin{enumerate}[leftmargin=1.9em, itemsep=0.38in]

  \item In this table, the \textbf{individuals} are \ans{the students renting instruments}.
        \vspace{10pt}

        Name one \textbf{variable} \ans{Instrument} and one of its \textbf{values}
        \ans{Clarinet}.

  \item Which of these lists could you add up and get something meaningful out of the
        total? \textbf{Circle} every one that works.
        \vspace{12pt}

        \begin{tabularx}{\linewidth}{@{\hspace{1.2em}} X X @{}}
        \ans{\textbf{(A)} the prices of five books in a cart} &
        \textbf{(B)} the numbers on five players' jerseys \\[13pt]
        \ans{\textbf{(C)} the ages of five people in a lobby} &
        \textbf{(D)} the house numbers on one street \\
        \end{tabularx}
        \vspace{16pt}

        What do the ones you circled have in common that the others do not?\\[10pt]
        \ansline{A and C record an amount --- how many dollars, how many years old.}\\[8pt]\ansline{B and D are names written in digits, so a total means nothing.}

  \item Your school ID card has a number printed on it. Is that number \emph{measuring}
        anything about you? If not, say what it is doing instead.\\[10pt]
        \ansline{No. It measures nothing --- two students with nearby ID numbers}\\[8pt]\ansline{are not alike in any way. The number names me, using digits.}

\end{enumerate}
