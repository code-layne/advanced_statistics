  \vspace{10pt}
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \begin{block}{Unimodal}
        One main peak.
      \end{block}
      \vspace{3pt}
      {\small Heights of one class $\cdot$ the sleep survey}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Bimodal}
        Two prominent peaks (EK UNC-1.H.4).
      \end{block}
      \vspace{3pt}
      {\small The volleyball club --- two groups in one picture}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Uniform}
        Every bar about the same height; no prominent peak.
      \end{block}
      \vspace{3pt}
      {\small Rolls of a fair die}
    \end{column}
  \end{columns}
  \vspace{14pt}
  \begin{center}
    {\small\color{slate} Peaks are part of \textbf{shape}. Say them in the same breath as
    ``symmetric'' or ``skewed.''}
  \end{center}
\end{frame}

% ── CENTER, SPREAD, UNUSUAL FEATURES ────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{The other three}
  \sectionlabel{center $\cdot$ spread $\cdot$ unusual features}
  \vspace{4pt}
  \begin{columns}[t]
    \begin{column}{0.47\textwidth}
      \begin{block}{Center}
        The typical value. \textbf{Line A: 4 minutes. Line B: 4 minutes.}
      \end{block}
      \vspace{4pt}
      \begin{block}{Variability (spread)}
        How far the values reach. \textbf{Line A: 3 to 6. Line B: 1 to 13.}
      \end{block}
      \vspace{4pt}
      {\small Same center, different spread \redink{$\Rightarrow$} different line.}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Unusual features}
        \begin{itemize}\setlength{\itemsep}{3pt}
          \item \textbf{Outlier} --- unusually small or large \emph{relative to the rest}
                (EK UNC-1.H.2). The 13-minute customer.
          \item \textbf{Gap} --- a stretch with nothing observed (EK UNC-1.H.5).
          \item \textbf{Cluster} --- a concentration of values, usually set off by gaps
                (EK UNC-1.H.6).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{6pt}
  \begin{center}
    {\small\color{slate} You do not delete an outlier for being inconvenient. You report it.}
  \end{center}
\end{frame}

% ── THE CRUX ─────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Point at the club member who is 67 inches tall}
  \sectionlabel{the whole point of today}
  \vspace{2pt}
  \begin{columns}[t]
    \begin{column}{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.85]
        \foreach \i/\c in {0/2, 1/5, 2/9, 3/4, 4/0, 5/0, 6/4, 7/9, 8/5, 9/2} {
          \fill[navy] ({\i*0.38},0) rectangle ({(\i+1)*0.38},{\c*0.16});
          \draw[white, very thin] ({\i*0.38},0) rectangle ({(\i+1)*0.38},{\c*0.16});}
        \draw[->, thick] (0,0) -- (0,2.0);
        \draw[->, thick] (0,0) -- (4.1,0);
        \foreach \x/\lab in {0/58, 1.52/66, 3.04/74} {
          \draw (\x,-0.07) -- (\x,0.07); \node[below, font=\tiny] at (\x,-0.1) {\lab};}
        \node[font=\tiny] at (1.9,-0.55) {Height (inches)};
      \end{tikzpicture}
      \end{center}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item A single ``typical height'' for this club lands around \textbf{67 inches}.
        \item \redink{Nobody in the club is 67 inches tall.} That stretch is empty.
        \item The club is two clusters --- one near 63 inches, one near 73 --- and one number
              describes \redink{neither}.
      \end{itemize}
      \vspace{6pt}
      \begin{block}{Essential Knowledge (UNC-1.H.1)}
        A description needs \textbf{shape, center, variability, and unusual features}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ── SOCS + THE UNC-1.H.7 LIMIT ──────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Four things or nothing}
  \sectionlabel[goldacc]{S O C S --- and every one in context}
  \vspace{8pt}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{S}hape --- symmetric or skewed (named for the tail); unimodal, bimodal,
              uniform.
        \item \textbf{O}utliers and other unusual features --- outliers, gaps, clusters.
        \item \textbf{C}enter --- the typical value.
        \item \textbf{S}pread --- how far the values reach.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{In context, or it earns nothing}
        Name the \textbf{variable} and its \textbf{units} every time. ``Skewed right, center
        about 6'' is not an answer. ``Skewed right, center about 6 grams of sugar per
        serving'' is.
      \end{block}
      \vspace{4pt}
      {\small\color{slate} And (EK UNC-1.H.7): describing \emph{these} data is not a claim
      about a larger population --- it can only suggest one to test later.}
    \end{column}
  \end{columns}
\end{frame}

% ── APPLICATION ──────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Application: The Sleep Survey}
  \sectionlabel[goldacc]{We work this one together --- you hold the pen}
  Thirty students, asked how many hours of sleep they got on a school night.
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item How many students? Which value sits apart from all the rest?
    \item Which way does it lean --- and what does that tail mean \emph{for these students}?
    \item Then the whole thing: all four parts, one paragraph, in context.
  \end{itemize}
  \begin{block}{The question behind the question}
    You will say ``skewed left.'' I will ask you to say it again \redink{about students},
    not about the graph.
  \end{block}
\end{frame}
