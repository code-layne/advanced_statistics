% GRADUAL RELEASE: the vocabulary is taught in the guided notes, so the targets name it

\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item \ldots{} explain why the center of a distribution, reported by itself, is not a
        description of anything.
  \item \ldots{} name the \textbf{shape} of a distribution --- symmetric, skewed left, or
        skewed right --- and say why the \emph{tail} decides that name, not the pile.
  \item \ldots{} point out the \textbf{unusual features} of a display: outliers, gaps, and
        clusters --- and say why an outlier gets investigated rather than deleted.
  \item \ldots{} describe a distribution completely --- \textbf{shape, outliers, center,
        spread} --- with every one of the four stated in context.
\end{enumerate}
\end{learningtargetbox}
