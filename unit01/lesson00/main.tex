\begin{tcolorbox}[colback=sky, colframe=navy, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students will explain that a number carries meaning only when
    placed in context, identify the individuals and variables in a data set, and distinguish a
    statistical question --- one that anticipates variability and is answered by collecting
    data --- from a question with a single definite answer.
    \\[2pt]
    \textbf{CED alignment:} Topic 1.1 \quad$\cdot$\quad Big Idea: VAR (Variation \& Distribution)
    \quad$\cdot$\quad Skill 1.A \quad$\cdot$\quad LO VAR-1.A \quad$\cdot$\quad EK VAR-1.A.1
    \\[2pt]
    \textbf{Lesson model:} \emph{Gradual release --- I do, we do, you do.} The guided notes
    deliver the vocabulary and the two checks directly; the guided practice is worked together
    with students holding the pen; the independent practice and the group activity are where
    students carry it alone and then with peers. The debrief closes the loop as a whole class.
\end{tcolorbox}

\boxguard
\begin{skillbox}[Learning Targets \& Key Understandings]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Learning targets (student language)} & \textbf{Key understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item I can explain why a bare number tells me nothing, and name the context it needs.
  \item I can identify the individuals and the variables in a data set, and tell a variable
        apart from one of its values.
  \item I can explain what variability is and why it is the reason statistics exists.
  \item I can decide whether a question is statistical, justify it by pointing to what varies,
        and write one of my own.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Numbers convey meaningful information only when placed in context (EK VAR-1.A.1).
  \item Variability is the reason statistics exists: if every individual gave the same value,
        one measurement would answer everything.
  \item \textbf{Target misconception:} that any question with a numerical answer, or any
        question needing the whole table, is statistical. \emph{``How many dogs are on the
        board?''} needs every row and returns a number --- and is \textbf{not} statistical,
        because its answer does not differ from one dog to the next. The discriminator is
        \emph{variation across individuals}, not arithmetic.
\end{itemize}
\end{tabularx}
\end{skillbox}

\boxguard
\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- taught directly in the guided notes}]{greenbox}
\begin{minipage}[t]{0.48\linewidth}
    \begin{itemize}
        \item \textbf{Statistics} --- the science of collecting, organizing, analyzing, and
              drawing conclusions from data
        \item \textbf{Data} --- recorded values, together with the context that gives them meaning
        \item \textbf{Context} --- who/what was measured, when, where, and in what units
              (EK VAR-1.A.1)
        \item \textbf{Individual} --- the person, animal, or object described by one row
    \end{itemize}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}
    \begin{itemize}
        \item \textbf{Variable} --- a characteristic recorded for each individual; a column
              heading, never one of the entries beneath it
        \item \textbf{Value} --- one entry under a variable
        \item \textbf{Variability} --- values differ across individuals; why one measurement
              is not enough
        \item \textbf{Statistical question} --- anticipates variability and is answered by
              collecting data
    \end{itemize}
\end{minipage}
\end{skillbox}

\begin{skillbox}[Warm-Up --- Activate Prior Knowledge (5 min)]{sky}
\begin{minipage}[t]{0.48\linewidth}
    \textbf{The three items and what each seeds}
    \begin{itemize}
        \item \textbf{1.} Seven numbers, no context: \emph{``What do these tell you?''} The
              intended answer is \emph{``I can't tell.''} \textbf{Seeds:} context / EK VAR-1.A.1.
        \item \textbf{2.} ``Three things you'd need to know.'' \textbf{Seeds:} the W questions ---
              their own list becomes the notes' W-question table.
        \item \textbf{3.} ``How tall is Maria?'' vs.\ ``How tall are the students in this class?''
              \textbf{Seeds:} the statistical-question test in notes section~4.
    \end{itemize}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}
    \textbf{Running it}
    \begin{itemize}
        \item \textbf{Resist rescuing them on item~1.} The discomfort is what makes the
              definition of \emph{data} land three minutes later. Give three minutes, then ask
              only \emph{``what would you need to know?''}
        \item Write their own phrasings on the board and leave them there --- you will circle
              them and write \emph{context} and \emph{W questions} beside them during the notes.
        \item Debrief item~3 aloud; it is the same discrimination the activity's crux tests.
        \item \emph{This is the course's first lesson --- there is no prior unit to spiral.}
    \end{itemize}
\end{minipage}
\end{skillbox}

\begin{skillbox}[Guided Notes \& Practice --- I do, we do, you do (22 min)]{sky}
\begin{multicols}{2}
\textbf{I do (about 12 min).} Present notes sections 1--4 in order and have students fill the
blanks as each idea is named. Fill the vocabulary box \emph{as you go}, not in advance --- a term
is written the moment it is used.

\emph{Section 1} lands on the warm-up numbers: put the W-question table on the board and let
their own list from item~2 populate it. \emph{Section 2} is the identification move; trace a row
and then a column with your finger on the projected table before students write.
\textbf{The red caution box is the day-one error} --- ``Yes'' is not a variable, \emph{Rides the
bus?} is. \emph{Section 3} is short and is the sentence students should leave with:
\textbf{variability is why statistics exists}. \emph{Section 4} gives the two checks, then the
S/N table. Work the first row aloud, then release rows 2--4.

\columnbreak
\textbf{We do (about 6 min).} The Guided Practice box, \emph{The Gym Sign-Up}. Students hold the
pen; you hold the questions: \emph{``Who is one individual here --- the gym, or a member?''}
\quad \emph{``Say your question again, but tell me what differs from member to member.''}
\quad \textbf{Part (c) is the point of the box:} ZIP code is a label, not a measurement, and
students who have quietly decided ``statistical = numeric'' will say no. Run both checks aloud
on it: \emph{``Does check~1 say the answers have to be numbers, or just that they have to
differ?''}

\textbf{You do (about 4 min).} Independent Practice, three items, silent and alone. Circulate.
\textbf{Item~2 is the formative check} --- glance at it on every desk. If more than a third
choose (i) or cannot name the failing check, reteach the two checks before releasing the
activity, and cut activity scenario~1 to items (a)--(b) to buy the time.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Group Activity --- \emph{The Adoption Board} (16 min)]{sky}
\textbf{One activity for the whole class --- there is no tiered version.} Groups of 3--4.
\textbf{Launch (1 min).} Project the board. \emph{``This is one shelter's adoption board.
Visitors read it, and so does the director --- but not for the same things. Use today's
vocabulary, and for every answer be ready to show me \textbf{where on the board} you see it.''}

\begin{multicols}{2}
\textbf{What students do.} \emph{Scenario 1} (about 6 min) applies the notes to a table they
have not seen: individual and count, variable-vs-value on ``Beagle'' (1b), a single value versus
an entire variable (1c), and variability as the reason a column is worth posting (1d).

\emph{Scenario 2} (about 10 min) is the contrast case. Item 2a sorts four director's questions
into ``one spot'' or ``many rows.'' \textbf{2b is the crux:} three of the four need many rows and
only one of those three is \emph{not} statistical. Items 2c--2d turn the flyer claim into the
summary-does-not-predict-an-individual idea.

\columnbreak
\textbf{What the teacher does.} Circulate and press for evidence from the board rather than
supplying answers:
\begin{itemize}[leftmargin=1.1em, nosep]
  \item \textbf{1b:} ``Point at where you see the word `Breed.' Now point at `Beagle.' Which one
        is the heading?''
  \item \textbf{1d:} ``Cover the Days column with your hand. What did you just lose?''
  \item \textbf{2b (the crux):} ``Write the answer to (B). Now write the answer to (C) --- for
        Sage, then for Ranger. How many answers did each question have?''
  \item \textbf{2d:} ``Find the shortest wait on the board. Now the longest. Is 25 either one?''
\end{itemize}
While circulating, pick the work to display: one board with 1c underlined well, and one group's
2b answer that is \emph{almost} right (many-rows $=$ statistical) to open the debrief with.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Debrief --- whole class (8 min)]{sky}
Groups report out, then fill the Debrief box at the end of the activity together.
\begin{multicols}{2}
\begin{enumerate}[leftmargin=1.3em, nosep, itemsep=3pt]
  \item \textbf{Open on the \emph{almost}-right 2b answer} you collected --- the group that
        decided many-rows means statistical. Correcting it in public, gently, is what makes the
        crux stick. Put (B), (C), and (D) side by side on the board and count answers for each.
  \item Take 1b from a group that got it: circle \emph{Breed} on the projected board, then
        circle \emph{Beagle}, and write \textbf{variable} and \textbf{value} beside them.
  \item Take 1d and say plainly: \textbf{this is why the course exists}. If the column did not
        vary, there would be nothing to post.
  \item Take 2d last. Ask for the shortest and longest wait out loud before anyone explains.
\end{enumerate}
\columnbreak
\textbf{The Debrief box.} Three questions, filled together as the report-out lands. Item~1 is
the sentence you want on every desk; item~2 forces a self-generated counterexample; item~3 is
the ``why this course exists'' line in their own words.

\textbf{Pose cold, if time allows:} \emph{``Is `What is the most common breed on this board?'
statistical?''} Breed is not a number and the answer is one word --- but it varies across dogs
and needs the data. It is the same special case as the Guided Practice's part (c), and asking it
twice in one period is what dislodges ``statistical $=$ numeric.''

\textbf{If the debrief runs short}, cut the cold-call and go straight to the homework launch ---
Debrief item~1 is the only one that cannot be cut.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[AP Practice --- assigned, not scored]{goldbox}
Four AP-style multiple-choice items (five options each, in context) on the adoption board and a
new veterinary context, then one multi-part free-response set on a school-newspaper survey:
identify individuals and variables, write a statistical question with what varies, critique a
bare claim, and explain why a group summary does not predict one individual. \textbf{No spiral item --- this is the course's first lesson.}

\textbf{Not scored --- the cover's score column reads \textbf{NA} for this page.} Go over it at
the start of Lesson~1.1 and hold the AP standard out loud: \emph{in context or it does not
count}. A part (b) that names a question but not what varies is incomplete; a part (d) that
restates the claim without mentioning that hours differ across seniors does not earn the point.
Students keep this page --- it is their AP-format rep, not a grade.
\end{skillbox}

\boxguard
\begin{skillbox}[Homework --- scored, due next class]{goldbox}
\emph{The Food Truck Lot}, a record students have not seen. Part~A is identification and one
student-written statistical question; Part~B is the contrast pair plus a prompt that forces a
\emph{non-numeric} statistical question; Part~C is the context critique and one AP-style
multiple-choice item with a required justification. The packet closes with the Lesson~1.1
preview.

\textbf{Start it in the last four minutes of the period} so students hit item~1 while you are
still in the room. \textbf{Scoring:} 10 points --- 2 for A1--A2 combined, 2 for A3
(the question must name what varies), 2 for B1, 2 for B2, 2 for C1--C2. \textbf{B2 is the
one to read closely}: a student who cannot produce a non-numeric statistical question has not
separated \emph{varies} from \emph{numeric}, and that is the misconception that follows them
into Lesson~1.1.

\textbf{DeltaMath override:} when DeltaMath carries this topic, assign it in place of this page,
strike through the score cell on the cover, and leave this page in the packet as extra practice.
\end{skillbox}

\boxguard
\begin{skillbox}[Watch For (while circulating)]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Value named where a variable belongs} (notes 2, activity 1b, HW~A2): ``Beagle''
        for ``Breed.'' Probe: \emph{``Is that the heading, or something written under it?''}
  \item \textbf{``Many rows'' equated with ``statistical''} (activity 2b --- the crux, HW~B1):
        counting the dogs needs the whole board and is still not statistical. Probe:
        \emph{``How many different answers does that question have?''}
  \item \textbf{``Has a number in it'' equated with ``statistical''} (HW~C2 distractors A, B, D).
        Probe: \emph{``Whose value? One individual's, or many?''}
  \item \textbf{``Statistical'' equated with ``numeric''} (Guided Practice~c, HW~B2). Probe:
        \emph{``Run check~1. Do members live in different ZIP codes?''}
  \item \textbf{Summary read as a prediction} (activity 2d, HW~C1, AP Practice~d): ``25 days''
        taken as a promise. Probe: \emph{``Show me a dog on the board that waited 25 days.''}
\end{itemize}
Cold-call prompts: \emph{``Give me one individual and one variable from your board.''} \quad
\emph{``Say a question that is NOT statistical, and tell me which check it fails.''}
\end{skillbox}

\boxguard
\begin{skillbox}[Close \& Assign (4 min)]{goldbox}
Hand the homework launch first --- \emph{The Food Truck Lot}, scored, due next class (or the
DeltaMath assignment in its place) --- and point out that the AP Practice page is theirs to work
and bring to review. Then close on what changed rather than on the assignment: \emph{``This
morning seven numbers meant nothing to you. You now know exactly what you were missing, and you
have a two-question test that tells you whether a question is even worth collecting data for.''}
\textbf{Preview:} Lesson~1.1, \emph{The Language of Variation} --- we split variables into
categorical and quantitative, the first decision that determines every graph and summary for the
rest of the course. Note that \emph{population} and \emph{sample} are deliberately \textbf{not}
introduced today; they belong to Unit~3, and naming them now competes with the
individual/variable distinction this lesson is built on.
\end{skillbox}

% ── Teacher notes — one per component, in packet order ────────────────────────
\begin{teachernote}[Warm-Up]
Deliberately unanswerable at first glance. Give three minutes and resist rescuing --- the
discomfort is what makes the definition of \emph{data} land in the notes. Expect some students to
guess ``test scores'' and defend it; validate the reasoning while pointing out that a guess is
not information. Debrief only item~3 aloud and leave it on the board all period. If it runs long,
cut item~2 and pick the ``what would you need to know'' list up during notes section~1 instead
--- but never cut item~1, which is the whole seed.
\end{teachernote}

\begin{teachernote}[Guided Notes \& Practice]
\textbf{Pace it 12 / 6 / 4.} The notes are four short sections, not a lecture: if you are talking
for more than 90 seconds without a student writing something, move. Sections 2 and 4 are the
must-land ones --- heading-vs-entry and the two checks. Section~3 is four sentences and should
stay that way; its whole job is the line \emph{variability is why statistics exists}.

The vocabulary box is filled \textbf{as each term is used}, never front-loaded. A student who
writes all five definitions before seeing an example has copied, not learned.

\textbf{Guided Practice is where the pen changes hands.} Do not work it on the board while
students watch; take answers, write what they say, and push back with the probes in the plan.
Part (c) is the point of the box --- if you only get through (a) and (c), that is the right trade.

\textbf{Independent Practice is the formative check.} Item~2 tells you whether the two checks
landed. Sort what you see into three piles as you circulate: circled (ii) with a
variation-based reason (ready); circled (ii) with a vague reason (ready, needs the language);
circled (i) (the misconception survived). If the third pile is more than a third of the room,
reteach the two checks now --- the group activity's crux will not work otherwise.
\end{teachernote}

\begin{teachernote}[Group Activity]
\textbf{16 minutes, roughly 6 on scenario~1 and 10 on scenario~2} --- scenario~2 carries the crux
and needs the time. Groups that finish scenario~1 early should be pushed to justify 1d out loud
rather than moving on. \textbf{One activity for the whole class:} do not split the room into
levels. The differentiation is in how you circulate --- shorter, more concrete probes at tables
that are stuck, and ``say that again using the word \emph{variability}'' at tables that are not.

The board is deliberately the same context students will meet again on the AP Practice page: the
context is already loaded, so the thinking stays on the distinctions rather than on re-reading a
table.

\textbf{Debrief (8 min).} Open on the \emph{almost}-right 2b answer you collected. The must-land
moments are the (B) / (C) / (D) comparison and Debrief item~1; everything else can be compressed.
If time is short, cut the cold-call breed question --- it is the same test as Guided Practice
part (c), which students have already run once.
\end{teachernote}

\begin{teachernote}[AP Practice]
\textbf{This page is not scored} --- the graded practice for this lesson is the homework that
follows it. Work it as AP-format reps and go over it at the start of Lesson~1.1. MC item~1
targets variable-vs-value, item~2 the crux, item~3 identification in a fresh context, and item~4
variability as the reason the subject exists. \textbf{Item~4 is the one worth reading closely}:
it is EK VAR-1.A.1 in its most transferable form and recurs on the AP exam as ``interpret in
context.'' On the free response, hold the line on context from day one --- part (d) is where
students first learn that an answer without the situation in it earns nothing. If you want an
enrichment hook, have students bring in a headline containing a number and strip it for the
context it leaves out; it pairs well with a bulletin-board display.
\end{teachernote}

\begin{teachernote}[Homework]
\textbf{Scored, 10 points, due next class.} Launch it in the last four minutes so students hit
Part~A while you can still answer. The food-truck lot is a fresh context on purpose: everything
transferred from the adoption board has to be transferred, not recalled.

\textbf{B2} --- ``give a question showing that a statistical question need not have a numerical
answer'' --- is the item that predicts Lesson~1.1 performance, because a student who cannot
separate \emph{varies} from \emph{numeric} will classify ZIP codes and jersey numbers as
quantitative next period. Read those first. \textbf{C1}'s second half (why the sign does not tell
one customer their wait) is the AP ``a summary does not predict an individual'' move; a
restatement of the sign without reference to the spread of waits does not earn it.

\textbf{When DeltaMath carries this topic, assign it instead}, strike the score cell on the
cover, and tell students this page is now optional practice. Do not assign both.
\end{teachernote}
