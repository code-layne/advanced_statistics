% GRADUAL RELEASE: the vocabulary is taught in the guided notes, so the targets name it
% plainly. The AP Learning Objective (VAR-1.A) stays in the teacher-facing lesson plan.
\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item \ldots{} explain why a number carries no information until it is placed in
        \textbf{context}, and name the context it needs.
  \item \ldots{} identify the \textbf{individuals}, \textbf{variables}, and \textbf{values}
        in a data set, and tell a variable apart from one of its values.
  \item \ldots{} explain what \textbf{variability} is and why it is the reason statistics
        exists.
  \item \ldots{} decide whether a question is a \textbf{statistical question}, and write one
        of my own about a given data set.
\end{enumerate}
\end{learningtargetbox}
