\begin{enumerate}[leftmargin=1.7em, itemsep=0.20in]

  \item \textbf{Circle} the statistical question.
        \begin{enumerate}[label=\textbf{(\Alph*)}, leftmargin=2em, itemsep=4pt]
          \item How many students are enrolled at this school?
          \item How many hours did Jordan work last week?
          \item How do the hours worked each week vary among students who have jobs?
                \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
          \item What time does the library close on Friday?
        \end{enumerate}

  \item A report states only: \textbf{``The average is 12.''} Name \textbf{two} pieces
        of context you would need before this number means anything.\\[4pt]
        \ansline{1. WHO or WHAT was measured --- the individuals behind the average.}\\[1pt]\ansline{2. WHAT was measured, and in what units. (Also acceptable: when and where.)}

  \item A biologist weighs each of the 30 turtles in a pond and records the weight in
        kilograms.
        \vspace{6pt}

        \textbf{(a)} The individuals are \ans{the 30 turtles in the pond}.

        \vspace{6pt}
        \textbf{(b)} The variable is \ans{weight}, measured in \ans{kilograms}.

        \vspace{8pt}
        \textbf{(c)} Write one statistical question about this data set, then name what
        varies.\\[4pt]
        \ansline{``How do the weights of the turtles in this pond vary?'' What varies is}\\[1pt]\ansline{the weight, in kilograms, from one turtle to the next.}

\end{enumerate}
