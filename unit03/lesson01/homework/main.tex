\begin{practicebox}
\small
For each survey below, (a) classify it as trustworthy or not trustworthy,
(b) identify whose opinion is over- or under-represented, and
(c) explain how the results might be distorted.

\vspace{8pt}
\textbf{1.} A magazine publishes an online poll asking readers whether they support
a new environmental regulation. Over 10{,}000 people vote.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Trustworthy / Not Trustworthy \quad \gradedhide{\writeline}\gradedright{\gradedblank[id=hw1a, type=short, points=1, answer={not trustworthy}, check=text]{1.6in}{0.2in}}
  \item Over/under-represented: \gradedhide{\writeline}\gradedright{\gradedblank[id=hw1b, type=rubric, points=2, display={Over: strong opinions. Under: neutral people, non-readers.}]{4.2in}{0.2in}}
  \item Distortion: \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw1c, type=rubric, points=2, display={Overstates strong views; 10,000 voluntary votes are still biased.}]{5in}{\gradedlinesheight{2}}}\hspace{\dimexpr 5in-3pt\relax}}\\[1pt]\writeline}
\end{enumerate}

\vspace{8pt}
\textbf{2.} A health researcher randomly selects 150 adults from a hospital's patient
database and asks about their exercise habits.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Trustworthy / Not Trustworthy \quad \gradedhide{\writeline}\gradedright{\gradedblank[id=hw2a, type=short, points=1, answer=trustworthy, check=text]{1.6in}{0.2in}}
  \item Who is included, and is there any concern? \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw2b, type=rubric, points=2, display={Hospital patients only; may not represent all adults (less healthy).}]{3.2in}{\gradedlinesheight{2}}}\hspace{\dimexpr 3.2in-3pt\relax}}\\[1pt]\writeline}
\end{enumerate}

\vspace{8pt}
\textbf{3.} A company surveys its own employees to determine what the average worker
in the city earns.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Trustworthy / Not Trustworthy \quad \gradedhide{\writeline}\gradedright{\gradedblank[id=hw3a, type=short, points=1, answer={not trustworthy}, check=text]{1.6in}{0.2in}}
  \item Whose opinion is missing? \gradedhide{\writeline}\gradedright{\gradedblank[id=hw3b, type=rubric, points=1, display={Workers at other companies, self-employed, other industries.}]{4.2in}{0.2in}}
  \item How might the result be distorted? \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw3c, type=rubric, points=2, display={This company's pay may be above or below the city's; estimate skewed.}]{3.7in}{\gradedlinesheight{2}}}\hspace{\dimexpr 3.7in-3pt\relax}}\\[1pt]\writeline}
\end{enumerate}

\vspace{8pt}
\textbf{4.} A researcher mails surveys to a random sample of 2{,}000 registered
voters. Only 300 respond.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Was the initial selection trustworthy? \makebox[3cm][r]{\smash{\gradedblank[id=hw4a, type=short, points=1, answer=yes, check=text]{0.55in}{0.2in}}}
  \item What concern arises from the low response rate? \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw4b, type=rubric, points=2, display={Nonresponse bias: the 300 responders may differ from the 1,700 who did not.}]{2.9in}{\gradedlinesheight{2}}}\hspace{\dimexpr 2.9in-3pt\relax}}\\[1pt]\writeline}
  \item Is a bigger (non-random) sample better? Explain. \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw4c, type=rubric, points=2, display={No: a bigger non-random sample is still biased; random is better.}]{2.9in}{\gradedlinesheight{2}}}\hspace{\dimexpr 2.9in-3pt\relax}}\\[1pt]\writeline}
\end{enumerate}
\end{practicebox}
