% ── ROW 1: ASK HOW ───────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Before you trust a number, ask how}
  \sectionlabel{notes, Ask How $\cdot$ problems 1--3}
  \vspace{2pt}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Data collection method}
        \textbf{How} the individuals in the data were chosen and measured.
      \end{block}
      \vspace{3pt}
      \begin{block}{Chance-based method}
        A \textbf{random} process decides who is in the data.
      \end{block}
      \vspace{3pt}
      \begin{block}{Non-chance method}
        \textbf{People} decide --- the researcher takes whoever is handy, or respondents
        volunteer.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\small\textbf{The four questions}}
      \begin{enumerate}\setlength{\itemsep}{3pt}\small
        \item Who is in the data?
        \item Who decided they were in it --- \redink{chance} or people?
        \item Who is left out, and who is over-represented?
        \item So can the result be trusted for \emph{everyone}?
      \end{enumerate}
      \vspace{6pt}
      {\small Radio call-in vote? A coach's varsity players? 60 IDs picked by computer?}
    \end{column}
  \end{columns}
\end{frame}

% ── ROW 2: WHO IS MISSING? ───────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Who is missing?}
  \sectionlabel{notes, Who Is Missing? $\cdot$ problems 4--7}
  \vspace{2pt}
  A method has \textbf{bias} if it \redink{consistently} overestimates or underestimates the
  truth. \quad Does the city support a skate park?
  \vspace{6pt}
  \begin{center}
  {\small\renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{} c l r @{}}
    & \textbf{Method} & \textbf{Support} \\ \hline
    A & Online poll on a local skate shop's website (900 votes) & 88\% \\
    B & 150 people at the public library, Tuesday at 10 a.m. & 34\% \\
    C & 400 households picked at random from the city's list & 52\% \\
  \end{tabular}}
  \end{center}
  \vspace{4pt}
  \begin{block}{Ask it every time}
    Who decided who answered? Who does that favor --- and which way does it push the number?
    \quad 88\% describes \redink{who chose to vote}, not the city.
  \end{block}
\end{frame}

% ── ROW 3: SAMPLE SIZE — THE CRUX ────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Big is not the same as trustworthy}
  \sectionlabel{notes, Sample Size $\cdot$ problems 8--10 $\cdot$ the whole point of today}
  \vspace{6pt}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{News-site poll}
        \textbf{2{,}400} students chose to respond.\\
        \textbf{85\%} want a later start.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Random sample}
        \textbf{80} students drawn at random from the roster.\\
        \textbf{58\%} want a later start.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{8pt}
  ``2{,}400 is 30 times as many as 80, so the 85\% is more trustworthy.'' \emph{Agree?}
  \vspace{8pt}
  \begin{block}{Say it, write it in the box}
    A bigger sample from a non-chance method is still \redink{biased}. More volunteers are just
    more of the same volunteers.
  \end{block}
\end{frame}

% ── WE DO ────────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Guided Practice: The Step Challenge}
  \sectionlabel[goldacc]{We do $\cdot$ you hold the pen $\cdot$ problems 11--14}
  A fitness app wants the average steps per day for \emph{all} its users.
  \vspace{4pt}
  \begin{center}
  {\small\renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{} l r @{}}
    \textbf{Method} & \textbf{Mean steps/day} \\ \hline
    Leaderboard: the 12{,}000 users who joined the step challenge & 11{,}200 \\
    Random: 500 accounts picked at random from all users & 6{,}900 \\
  \end{tabular}}
  \end{center}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Which method used chance? Who decided in the other?
    \item Who is over-represented on the leaderboard? Is 11{,}200 too high or too low?
    \item ``24 times as many users, so report 11{,}200.'' Respond.
  \end{itemize}
\end{frame}

% ── DEBRIEF ──────────────────────────────────────────────────────────────────
\begin{frame}[t, plain]
  \navyheader{Debrief: who, not how many}
  \sectionlabel{8 minutes $\cdot$ pens down, papers up --- we say these aloud}
  \vspace{6pt}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{People decided}
        2{,}400 volunteers: \textbf{85\%}.\\
        12{,}000 on a leaderboard: \textbf{11{,}200} steps.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Chance decided}
        80 at random: \textbf{58\%}.\\
        500 at random: \textbf{6{,}900} steps.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{12pt}
  \begin{block}{Two things to say out loud}
    Why is ``80 is plenty'' the \emph{wrong} reason to trust the random sample? And did the
    skate-shop poll tell us about the city, or about \redink{who chose to vote}?
  \end{block}
\end{frame}
