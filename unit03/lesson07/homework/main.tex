\textbf{1.} 800 adults are randomly selected from a national voter registry and
surveyed about their political views and social media use. Heavier social media users
report more polarized political views.

\vspace{4pt}

\rule{0pt}{0.3in}Generalize to? \makebox[8cm][l]{\vphantom{\blank{8cm}}\hspace{46bp}\smash{\gradedblank[id=hw1d, type=short, points=1, answer={all national registered voters}, check=text, display={\bfseries all national registered voters}]{3.6in}{0.3in}}}\\[14pt]

\textbf{2.} 60 student volunteers are randomly assigned to either an active-learning

\vspace{4pt}
Random assignment? \makebox[2cm][l]{\vphantom{\blank{2cm}}\hspace{43bp}\smash{\gradedblank[id=hw2a, type=short, points=1, answer=yes, check=text]{0.33in}{0.17in}}} \quad Random selection? \makebox[2cm][l]{\vphantom{\blank{2cm}}\hspace{46bp}\smash{\gradedblank[id=hw2b, type=short, points=1, answer=no, check=text]{0.6in}{0.17in}}}\\[4pt]
Causation? \makebox[2cm][l]{\vphantom{\blank{2cm}}\hspace{43bp}\smash{\gradedblank[id=hw2c, type=short, points=1, answer=yes, check=text]{0.33in}{0.17in}}}\\[4pt]
\rule{0pt}{0.3in}Generalize to? \makebox[8cm][l]{\vphantom{\blank{8cm}}\hspace{46bp}\smash{\gradedblank[id=hw2d, type=short, points=1, answer={similar volunteers only}, check=text, display={\bfseries similar volunteers only}]{3.6in}{0.3in}}}\\[14pt]

\gradedhide{\gradedhere{\gradedblank[id=hw2e, type=rubric, points=2, display={Random assignment: active learning caused the higher scores. Volunteers: generalize only to students like them.}]{\linewidth+6pt}{2in}}\writelines{9}}
