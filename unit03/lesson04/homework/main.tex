\begin{practicebox}
\small
For each survey, (a) identify the type of bias, (b) explain how it distorts the
estimate (direction), and (c) suggest an improvement.

\vspace{8pt}
\textbf{1.} A radio station asks listeners to call in and answer: ``Do you think
schools should start later?'' 78\% of 2{,}400 callers say yes.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=7pt]
  \item Type of bias: \makebox[8cm][l]{\vphantom{\blank{8cm}}\hspace{46bp}\smash{\gradedblank[id=hw1a, type=short, points=1, answer={voluntary response bias}, check=text]{2.4in}{0.2in}}}
  \item Direction and explanation: \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw1b, type=rubric, points=2, display={Too high: strong opinions (mostly in favor) call in; neutral or opposing views are underrepresented.}]{4.2in}{\gradedlinesheight{2}}}\hspace{\dimexpr 4.2in-3pt\relax}}\\[1pt]\writelines{9}}
  \item Improvement: \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw1c, type=rubric, points=1, display={Use a random sample of parents and students from school records.}]{4.8in}{0.2in}}\hspace{\dimexpr 4.8in-3pt\relax}}\\[1pt]\writelines{9}}
\end{enumerate}

\vspace{8pt}
\textbf{2.} A phone survey randomly selects households from a list of landline
numbers to ask about online shopping habits.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=7pt]
  \item Type of bias: \makebox[8cm][l]{\vphantom{\blank{8cm}}\hspace{46bp}\smash{\gradedblank[id=hw2a, type=short, points=1, answer={undercoverage bias}, check=text]{2.4in}{0.2in}}}
  \item Who is excluded? \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw2b, type=rubric, points=1, display={Households without landlines (younger, lower-income).}]{4.5in}{0.2in}}\hspace{\dimexpr 4.5in-3pt\relax}}\\[1pt]\writelines{9}}
  \item How does this distort the estimate of online shopping frequency?
        \gradedhide{\writeline\\[1pt]\writeline\gradedright{\gradedhere{\gradedblank[id=hw2c, type=rubric, points=2, display={Too low: the excluded younger adults shop online more often.}]{6in}{0.8\baselineskip+12pt}}\hspace{\dimexpr 6in-3pt\relax}}\\[1pt]\writelines{9}}
\end{enumerate}

\vspace{8pt}
\textbf{3.} Researchers ask college students: ``Don't you think the university
should invest more in mental health resources?''

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=7pt]
  \item Type of bias: \makebox[8cm][l]{\vphantom{\blank{8cm}}\hspace{46bp}\smash{\gradedblank[id=hw3a, type=short, points=1, answer={response bias}, check=text]{2.4in}{0.2in}}}
  \item Rewrite the question to remove the bias: \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw3b, type=rubric, points=1, display={Neutral wording, without ``Don't you think''.}]{3.3in}{0.2in}}\hspace{\dimexpr 3.3in-3pt\relax}}\\[1pt]\writelines{9}}
  \item Direction: the estimate of support will be too \makebox[3cm][l]{\vphantom{\blank{3cm}}\hspace{30bp}\smash{\gradedblank[id=hw3c1, type=short, points=1, answer=high, check=text]{0.85in}{0.2in}}} because\\[6pt]
        \gradedhide{\gradedhere{\gradedblank[id=hw3c2, type=rubric, points=1, display={``don't you think'' implies agreeing is expected.}]{\linewidth+6pt}{\gradedlinesheight{10}}}\writelines{10}}
\end{enumerate}

\tcbbreak
\vspace{8pt}
\textbf{4.} A researcher randomly selects 800 adults but only 200 complete the survey.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=7pt]
  \item Was the initial selection biased? \makebox[3cm][l]{\vphantom{\blank{3cm}}\hspace{46bp}\smash{\gradedblank[id=hw4a, type=short, points=1, answer=no, check=text]{0.9in}{0.2in}}}
  \item What bias affects the completed surveys? \makebox[6cm][l]{\vphantom{\blank{6cm}}\hspace{46bp}\smash{\gradedblank[id=hw4b, type=short, points=1, answer={nonresponse bias}, check=text]{1.8in}{0.2in}}}
  \item How could this distort the results? \gradedhide{\writeline\gradedright{\gradedhere{\gradedblank[id=hw4c, type=rubric, points=2, display={The 200 who responded may differ (e.g., more engaged); the estimate is pulled their way.}]{3.7in}{\gradedlinesheight{2}}}\hspace{\dimexpr 3.7in-3pt\relax}}\\[1pt]\writelines{9}}
\end{enumerate}
\end{practicebox}
