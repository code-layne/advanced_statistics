\textbf{1.} A bag contains 5 red, 4 blue, and 1 yellow marble. One marble is drawn
at random.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item How many total outcomes are in the sample space? \makebox[2cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1a, type=short, points=1, answer=10, check=numeric, tolerance=0]{1.8cm}{0.2in}}}}
  \item Calculate each probability:
        $P(\text{red}) = \makebox[2cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1b1, type=short, points=1, answer=0.5, check=numeric, tolerance=0.001]{0.9cm}{0.2in}}}}$ \qquad
        $P(\text{blue}) = \makebox[2cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1b2, type=short, points=1, answer=0.4, check=numeric, tolerance=0.001]{0.9cm}{0.2in}}}}$ \qquad
        $P(\text{yellow}) = \makebox[2cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1b3, type=short, points=1, answer=0.1, check=numeric, tolerance=0.001]{0.75cm}{0.2in}}}}$
  \item Verify: do the three probabilities sum to 1? Show the sum. \makebox[4cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1c, type=short, points=1, answer=1, check=numeric, tolerance=0, display={sum $= 1$}]{2.4cm}{0.2in}}}}
  \item $P(\text{not blue}) = \mbox{\hspace{2cm}}$ using the complement rule.\hfill\gradedright{\smash{\raisebox{-4pt}{\gradedblank[id=hw1d, type=short, points=1, answer=0.6, check=numeric, tolerance=0.001]{1.8cm}{0.2in}}}}
  \item Interpret $P(\text{red}) = 0.5$ in a sentence using ``in the long run.''\\
        \gradedhide{\gradedhere{\kern0.4in\gradedblank[id=hw1e, type=rubric, points=2, display={Over many draws with replacement, about half are red.}]{\linewidth+6pt-0.4in}{\gradedlinesheight{1}+8pt}}\writelines{10}}
\end{enumerate}

\textbf{2.} A health clinic records the blood type of each patient. In the past, the
relative frequencies have been: O = 0.44, A = 0.42, B = 0.10, AB = 0.04.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Verify these relative frequencies sum to 1. \makebox[5cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2a, type=short, points=1, answer=1, check=numeric, tolerance=0, display={sum $= 1$}]{3.4cm}{0.2in}}}}
  \item A patient is chosen at random. Calculate:
        $P(\text{not type O}) = \makebox[3cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2b1, type=short, points=1, answer=0.56, check=numeric, tolerance=0.001]{1.5cm}{0.2in}}}}$ \quad $P(\text{type A or type B}) = \makebox[3cm][l]{\hspace{0.7cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2b2, type=short, points=1, answer=0.52, check=numeric, tolerance=0.001]{2.3cm}{0.2in}}}}$
  \item Interpret $P(\text{type O}) = 0.44$ in the context of this clinic.\\
        \gradedhide{\gradedhere{\kern0.4in\gradedblank[id=hw2c, type=rubric, points=2, display={Over many randomly selected patients, about 44\% have type O.}]{\linewidth+6pt-0.4in}{\gradedlinesheight{1}+8pt}}\writelines{10}}
\end{enumerate}

\textbf{3.} A standard six-sided die is rolled twice. Use the $6 \times 6$ sample
space (36 equally likely outcomes).

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item $P(\text{both dice show the same number}) = \makebox[3cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw3a, type=short, points=1, answer=0.1667, check=numeric, tolerance=0.001, display={$1/6$}]{2cm}{0.2in}}}}$
  \item $P(\text{sum equals 2 or 12}) = \makebox[3cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw3b, type=short, points=1, answer=0.0556, check=numeric, tolerance=0.001, display={$1/18$}]{2cm}{0.2in}}}}$
  \item $P(\text{sum is NOT 2 or 12}) = \makebox[3cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw3c, type=short, points=1, answer=0.9444, check=numeric, tolerance=0.001, display={$17/18$}]{2cm}{0.2in}}}}$
  \item Is $P(\text{sum} = 7) + P(\text{sum} \ne 7) = 1$? Verify using your answers
        from the guided notes. \makebox[5cm][l]{\hspace{0.7cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw3d, type=short, points=1, answer=1, check=numeric, tolerance=0, display={$1/6 + 5/6 = 1$}]{4cm}{0.2in}}}}
\end{enumerate}

\textbf{5.} (AP-Style Multiple Choice) A spinner has 5 equal sections: 2 red,
2 blue, 1 green. Which is the probability of NOT spinning red?

\begin{enumerate}[label=(\Alph*), leftmargin=2em, itemsep=4pt]
  \item $2/5$
  \item $2/3$
  \item $3/5$
  \item $1/5$
        \hfill\gradedright{\textit{Letter:}~\gradedblank[id=hw5, type=choice, points=1, answer=C]{0.4in}{0.2in}}\rule[-14pt]{0pt}{32pt}
\end{enumerate}
