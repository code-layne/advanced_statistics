\begin{practicebox}
\small
A hospital recorded blood type and a health outcome (whether the patient
was diagnosed with a particular condition) for a sample of 240 patients.

\vspace{6pt}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\linewidth}{@{} l X X X X r @{}}
 & \textbf{Type A} & \textbf{Type B} & \textbf{Type AB} & \textbf{Type O} & \textbf{Total} \\
\hline
\textbf{Condition present}  & 30 & 20 & 10 & 40 & 100 \\
\textbf{Condition absent}   & 50 & 30 & 20 & 40 & 140 \\
\hline
\textbf{Total}              & 80 & 50 & 30 & 80 & 240 \\
\end{tabularx}

\vspace{8pt}
One patient is selected at random from this sample.

\vspace{8pt}
\textbf{1.} Let $A = \{\text{patient has Type A blood}\}$ and
$B = \{\text{patient has Type O blood}\}$.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=6pt]
  \item Are events $A$ and $B$ mutually exclusive? Justify in context.\\
        \gradedhide{\gradedhere{\kern\dimexpr\linewidth-4in+3pt\relax\gradedblank[id=hw1a, type=rubric, points=2, display={Yes: each patient has exactly one blood type, so no one is both Type A and Type O.}]{4in}{\gradedlinesheight{2}}}\writelines{10}}
  \item Find $P(A \cap B)$. \makebox[4cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1b, type=short, points=1, answer=0, check=numeric, tolerance=0]{2.4cm}{0.2in}}}}
  \item Find $P(A)$ and $P(B)$. \makebox[6cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{$P(A) =$~\gradedblank[id=hw1c1, type=short, points=1, answer=0.333, check=numeric, tolerance=0.002, display={$1/3$}]{1.6cm}{0.2in}\quad $P(B) =$~\gradedblank[id=hw1c2, type=short, points=1, answer=0.333, check=numeric, tolerance=0.002, display={$1/3$}]{1.6cm}{0.2in}}}}
\end{enumerate}

\vspace{8pt}
\textbf{2.} Let $C = \{\text{condition present}\}$ and $D = \{\text{Type B blood}\}$.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=6pt]
  \item Are events $C$ and $D$ mutually exclusive? Justify in context.\\
        \gradedhide{\gradedhere{\kern0.4in\gradedblank[id=hw2a, type=rubric, points=2, display={No: 20 patients have Type B blood and the condition, so both can occur.}]{\linewidth+6pt-0.4in}{\gradedlinesheight{2}}}\writelines{10}}
  \item Find $P(C \cap D)$. Show your work. \makebox[5cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2b, type=short, points=1, answer=0.0833, check=numeric, tolerance=0.002, display={$20/240 \approx 0.083$}]{3.4cm}{0.2in}}}}
\end{enumerate}

\tcbbreak
\vspace{8pt}
\textbf{3.} Let $E = \{\text{condition present}\}$ and $F = \{\text{condition absent}\}$.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=6pt]
  \item Are $E$ and $F$ mutually exclusive? Why or why not?\\
        \gradedhide{\gradedhere{\kern0.4in\gradedblank[id=hw3a, type=rubric, points=2, display={Yes: a patient cannot both have and not have the condition.}]{\linewidth+6pt-0.4in}{\gradedlinesheight{2}}}\writelines{10}}
  \item What is $P(E \cap F)$? \makebox[4cm][l]{\hspace{1.6cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw3b, type=short, points=1, answer=0, check=numeric, tolerance=0]{2.4cm}{0.2in}}}}
\end{enumerate}

\vspace{8pt}
\textbf{4.} (AP Free-Response Style) A student claims: ``Type A blood and having the
condition present are mutually exclusive events because not everyone with Type A blood
has the condition.''
\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=6pt]
  \item Is this claim correct? Use the table to support your answer.\\
        \gradedhide{\gradedhere{\kern\dimexpr\linewidth-4in+3pt\relax\gradedblank[id=hw4a, type=rubric, points=2, display={Incorrect: 30 patients have Type A blood and the condition, so the events overlap.}]{4in}{\gradedlinesheight{3}}}\writelines{10}}
  \item Write a correct complete-sentence explanation of whether these events are
        mutually exclusive.\\
        \gradedhide{\gradedhere{\kern\dimexpr\linewidth-4in+3pt\relax\gradedblank[id=hw4b, type=rubric, points=2, display={Not mutually exclusive: 30 patients in the sample have both Type A blood and the condition.}]{4in}{\gradedlinesheight{2}}}\writelines{10}}
\end{enumerate}

\tcbbreak
\vspace{8pt}
\textbf{5.} (Extension) Suppose a new event $G = \{\text{Type AB blood}\}$ is defined.
Can you find \emph{any} blood type event that is mutually exclusive with $G$?
How many can you find? Explain.\\
