\textbf{1.} A spinner is divided into 5 equal sections numbered 1--5. Let $X$ be the number
the spinner lands on.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Write the probability distribution for $X$ as a table.\gradedbelow[r]{0.03in}{\gradedblank[id=hw1a, type=rubric, points=1, display={$x = 1,2,3,4,5$; each $P(X=x) = 0.20$}]{4in}{0.24in}}

        \vspace{0.35in}

  \item Verify the two conditions for a valid probability distribution.\\
        \writeline\gradedhide{\hrulefill}\gradedright{\raisebox{-2pt}{\gradedblank[id=hw1b, type=rubric, points=2, display={All $0.20 \ge 0$; sum $= 1$.}]{2.4in}{0.22in}}}

  \item Find $P(X \ge 3)$.  \quad \makebox[4cm][l]{\hspace{1.3cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1c, type=short, points=1, answer=0.6, check=numeric, tolerance=0.001]{2.4cm}{0.2in}}}}

  \item Find $P(2 \le X \le 4)$. \quad \makebox[4cm][l]{\hspace{1.3cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw1d, type=short, points=1, answer=0.6, check=numeric, tolerance=0.001]{2.4cm}{0.2in}}}}

\textbf{2.} The table below shows the probability distribution for $Y$ = number of
broken eggs in a randomly selected carton of 12.

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}cccccc@{}}
$y$ & 0 & 1 & 2 & 3 & 4 \\
\hline
$P(Y=y)$ & 0.68 & 0.17 & 0.09 & ? & 0.02 \\
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=8pt]
  \item Find the missing probability $P(Y=3)$.  \quad \makebox[4cm][l]{\hspace{1.3cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2a, type=short, points=1, answer=0.04, check=numeric, tolerance=0.001]{2.4cm}{0.2in}}}}
  \item Find $P(Y \le 1)$.  \quad \makebox[4cm][l]{\hspace{1.3cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2b, type=short, points=1, answer=0.85, check=numeric, tolerance=0.001]{2.4cm}{0.2in}}}}
  \item Find $P(Y \ge 1)$.  \quad \makebox[4cm][l]{\hspace{1.3cm}\smash{\raisebox{-4pt}{\gradedblank[id=hw2c, type=short, points=1, answer=0.32, check=numeric, tolerance=0.001]{2.4cm}{0.2in}}}}
  \item Interpret $P(Y = 0) = 0.68$ in context.\\
        \gradedhide{\gradedhere{\kern0.4in\gradedblank[id=hw2d, type=rubric, points=2, display={There is a 68\% probability that a randomly selected carton has no broken eggs.}]{\linewidth+6pt-0.4in}{\gradedlinesheight{2}-5pt}}\writeline\\[1pt]\writeline}
\end{enumerate}
