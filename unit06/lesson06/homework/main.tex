\textit{Perform a complete four-step hypothesis test (State, Plan, Do, Conclude) for each problem.}

\begin{enumerate}

  \item A marketing firm claims 40\% of adults in a city have visited a particular
        website in the past month. A competitor believes the proportion is \emph{lower}.
        From a random sample of 350 adults, 126 have visited the site. Use $\alpha=0.05$.

        \textbf{State:} \gradedhide{\vspace{1pt}\noindent\gradedhere[0.17in-3pt]{\gradedblank[id=hw1a, type=rubric, points=2, display={\bfseries $H_0\colon p = 0.40$; $H_a\colon p < 0.40$}]{2.85in}{0.17in}}\textcolor{linegray}{\hrulefill}\vspace{3pt}}
        \textbf{Plan:} \gradedhide{\vspace{1pt}\noindent\gradedhere[0.17in-3pt]{\gradedblank[id=hw1b, type=rubric, points=3, display={\bfseries Random; $140 \ge 10$, $210 \ge 10$; 10\% met}]{2.85in}{0.17in}}\textcolor{linegray}{\hrulefill}\vspace{3pt}\\\writeline}
        \textbf{Do:} $\hat p = $ \rlap{\smash{\gradedblank[id=hw1c1, type=short, points=1, answer=0.36, check=numeric]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $z \approx$ \rlap{\smash{\gradedblank[id=hw1c2, type=short, points=2, answer=-1.53, check=numeric, tolerance=0.02]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $p$-value $\approx$ \rlap{\smash{\gradedblank[id=hw1c3, type=short, points=2, answer=0.063, check=numeric, tolerance=0.003]{2cm}{0.17in}}}\phantom{\blank{2cm}}
        \textbf{Conclude:} \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw1d, type=rubric, points=2, display={$0.063 > 0.05$: fail to reject $H_0$; no convincing evidence fewer than 40\% of the city's adults visited the site.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{1}}

  \item A national study reports that 72\% of high school students participate in at
        least one extracurricular activity. A district administrator believes the rate
        in her district has \emph{changed}. From 250 randomly selected students, 193
        participate. Use $\alpha=0.05$.

        \textbf{State:} \gradedhide{\vspace{1pt}\noindent\gradedhere[0.17in-3pt]{\gradedblank[id=hw2a, type=rubric, points=2, display={\bfseries $H_0\colon p = 0.72$; $H_a\colon p \ne 0.72$}]{2.85in}{0.17in}}\textcolor{linegray}{\hrulefill}\vspace{3pt}}
        \textbf{Plan:} \gradedhide{\vspace{1pt}\noindent\gradedhere[0.17in-3pt]{\gradedblank[id=hw2b, type=rubric, points=3, display={\bfseries Random; $180 \ge 10$, $70 \ge 10$; 10\% met}]{2.85in}{0.17in}}\textcolor{linegray}{\hrulefill}\vspace{3pt}\\\writeline}
        \textbf{Do:} $\hat p = $ \rlap{\smash{\gradedblank[id=hw2c1, type=short, points=1, answer=0.772, check=numeric]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $z \approx$ \rlap{\smash{\gradedblank[id=hw2c2, type=short, points=2, answer=1.83, check=numeric, tolerance=0.03]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $p$-value $\approx$ \rlap{\smash{\gradedblank[id=hw2c3, type=short, points=2, answer=0.067, check=numeric, tolerance=0.004]{2cm}{0.17in}}}\phantom{\blank{2cm}}
        \textbf{Conclude:} \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2d, type=rubric, points=2, display={$0.067 > 0.05$: fail to reject $H_0$; no convincing evidence the district's participation rate differs from 72\%.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{1}}

  \item A bottled water company claims fewer than 5\% of their bottles have a
        contamination level above the safe threshold. An environmental group tests 400
        randomly selected bottles; 26 exceed the threshold. Use $\alpha=0.01$.

        \textbf{State:} \gradedhide{\vspace{1pt}\noindent\gradedhere[0.17in-3pt]{\gradedblank[id=hw3a, type=rubric, points=2, display={\bfseries $H_0\colon p = 0.05$; $H_a\colon p > 0.05$}]{2.85in}{0.17in}}\textcolor{linegray}{\hrulefill}\vspace{3pt}}
        \textbf{Plan:} \gradedhide{\vspace{1pt}\noindent\gradedhere[0.17in-3pt]{\gradedblank[id=hw3b, type=rubric, points=3, display={\bfseries Random; $20 \ge 10$, $380 \ge 10$; 10\% met}]{2.85in}{0.17in}}\textcolor{linegray}{\hrulefill}\vspace{3pt}\\\writeline}
        \textbf{Do:} $\hat p = $ \rlap{\smash{\gradedblank[id=hw3c1, type=short, points=1, answer=0.065, check=numeric]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $z \approx$ \rlap{\smash{\gradedblank[id=hw3c2, type=short, points=2, answer=1.38, check=numeric, tolerance=0.02]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $p$-value $\approx$ \rlap{\smash{\gradedblank[id=hw3c3, type=short, points=2, answer=0.084, check=numeric, tolerance=0.003]{2cm}{0.17in}}}\phantom{\blank{2cm}}
        \textbf{Conclude:} \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw3d, type=rubric, points=2, display={$0.084 > 0.01$: fail to reject $H_0$; no convincing evidence more than 5\% of the bottles exceed the threshold.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{1}}

\end{enumerate}

\begin{reflectionbox}
\textbf{Preview:} Lesson 6.7 introduces Type I errors, Type II errors, and power. Reflect:
when you fail to reject $H_0$, does that mean $H_0$ is definitely true? What might be
happening if you keep getting $p > \alpha$ even when $H_a$ is actually true?
\end{reflectionbox}
