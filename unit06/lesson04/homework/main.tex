\textit{For each problem, state $H_0$ and $H_a$, verify conditions, and compute $z$.}

\begin{enumerate}

  \item The CDC reports 18\% of U.S.\ adults smoke cigarettes. A public health researcher
        wants to test whether the smoking rate in a particular county differs from the national
        rate. She randomly surveys 450 county adults and finds 72 are smokers.
        \begin{enumerate}[label=\alph*.]
          \item $H_0$: \rlap{\smash{\gradedblank[id=hw1a1, type=short, points=1, answer={p = 0.18}, check=text, display={\boldmath$p = 0.18$}]{5cm}{0.17in}}}\phantom{\blank{5cm}} \quad $H_a$: \rlap{\smash{\gradedblank[id=hw1a2, type=short, points=1, answer={p != 0.18}, check=text, display={\boldmath$p \ne 0.18$}]{4.6cm}{0.17in}}}\phantom{\blank{5cm}} \quad Type: \rlap{\smash{\gradedblank[id=hw1a3, type=short, points=1, answer=two-sided, check=text]{2cm}{0.17in}}}\phantom{\blank{2cm}}
          \item Conditions: \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw1b, type=rubric, points=3, display={Random sample; $450(0.18)=81 \ge 10$ and $450(0.82)=369 \ge 10$; $450 \le 10\%$ of county adults. All met.}]{\linewidth+6pt}{\gradedlinesheight{2}}}\writelines{9}}
          \item $\hat p = $ \rlap{\smash{\gradedblank[id=hw1c1, type=short, points=1, answer=0.16, check=numeric]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $z = $ \rlap{\smash{\gradedblank[id=hw1c2, type=short, points=1, answer=(0.16-0.18)/sqrt(0.18*0.82/450), check=expression, display={\boldmath$(0.16-0.18)/0.0181$}]{6cm}{0.17in}}}\phantom{\blank{6cm}} $\approx$ \rlap{\smash{\gradedblank[id=hw1c3, type=short, points=2, answer=-1.10, check=numeric, tolerance=0.02]{2cm}{0.17in}}}\phantom{\blank{2cm}}
        \end{enumerate}

  \newpage
  \item A seed company claims that 85\% of its seeds germinate. An agricultural researcher
        believes the true germination rate is \emph{lower} than claimed. She plants 200
        randomly selected seeds and 162 germinate.
        \begin{enumerate}[label=\alph*.]
          \item State hypotheses and justify the direction of $H_a$. \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2a, type=rubric, points=2, display={$H_0\colon p = 0.85$, $H_a\colon p < 0.85$: the researcher suspects the rate is lower than claimed.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}
          \item Verify conditions. \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2b, type=rubric, points=3, display={Random sample of seeds; $200(0.85)=170 \ge 10$ and $200(0.15)=30 \ge 10$; $200 \le 10\%$ of all seeds. All met.}]{\linewidth+6pt}{\gradedlinesheight{2}}}\writelines{9}}
          \item $\hat p = $ \rlap{\smash{\gradedblank[id=hw2c1, type=short, points=1, answer=0.81, check=numeric]{2cm}{0.17in}}}\phantom{\blank{2cm}} \quad $z \approx$ \rlap{\smash{\gradedblank[id=hw2c2, type=short, points=2, answer=-1.58, check=numeric, tolerance=0.02]{2cm}{0.17in}}}\phantom{\blank{2cm}}
        \end{enumerate}

  \newpage
  \item A school claims that 70\% of students participate in extracurricular activities.
        A parent group believes participation has \emph{increased}. A random sample of
        180 students finds 135 participate.
        \begin{enumerate}[label=\alph*.]
          \item Hypotheses: \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw3a, type=rubric, points=2, display={\bfseries $H_0\colon p = 0.70$; $H_a\colon p > 0.70$}]{\linewidth+6pt}{\gradedlinesheight{9}}}\writelines{9}}
          \item Conditions: \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw3b, type=rubric, points=3, display={Random; $180(0.70)=126 \ge 10$, $180(0.30)=54 \ge 10$; $180 \le 10\%$ of students. All met.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}
          \item $z \approx$ \rlap{\smash{\gradedblank[id=hw3c, type=short, points=2, answer=1.46, check=numeric, tolerance=0.02]{2cm}{0.17in}}}\phantom{\blank{2cm}}
        \end{enumerate}

  \item \textbf{Reflection:} In Problem 1, what changes in the setup if the researcher
        instead believed the county rate was \emph{higher} than 18\%? \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw4, type=rubric, points=2, display={$H_a$ becomes $p > 0.18$ (one-sided); $H_0$ and the conditions stay the same.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{9}}

\end{enumerate}

\begin{reflectionbox}
\textbf{Preview:} In Lesson 6.5 you will use the test statistic $z$ to find the $p$-value
and interpret it as evidence against $H_0$. The $p$-value answers: ``If $H_0$ were true,
how likely is a sample result at least this extreme?''
\end{reflectionbox}
