\textit{For each problem, state $H_0$ and $H_a$, verify conditions, and compute $z$.}

\begin{enumerate}

  \item CDC smoking rate test:
        \begin{enumerate}[label=\alph*.]
          \item $H_0$: \ans{$p=0.18$} \quad $H_a$: \ans{$p\ne0.18$} \quad Type: \ans{two-sided}\gradedkey{\gradedblank[id=hw1a1, type=short, points=1, answer={p = 0.18}, check=text, display={\boldmath$p = 0.18$}]{5cm}{0.17in}\quad \gradedblank[id=hw1a2, type=short, points=1, answer={p != 0.18}, check=text, display={\boldmath$p \ne 0.18$}]{4.6cm}{0.17in}\quad \gradedblank[id=hw1a3, type=short, points=1, answer=two-sided, check=text]{2cm}{0.17in}}
          \item \ansline{Random (SRS --- met); Large Counts: $450(0.18)=81\ge10$, $450(0.82)=369\ge10$ --- met; 10\% met.}\gradedkeyline{\gradedblank[id=hw1b, type=rubric, points=3, display={Random sample; $450(0.18)=81 \ge 10$ and $450(0.82)=369 \ge 10$; $450 \le 10\%$ of county adults. All met.}]{\linewidth+6pt}{\gradedlinesheight{2}}}
          \item $\hat p = $ \ans{$72/450=0.16$} \quad $z = \dfrac{0.16-0.18}{\sqrt{0.18(0.82)/450}} = \dfrac{-0.02}{0.01810} \approx$ \ans{$-1.11$}\gradedkey{\gradedblank[id=hw1c1, type=short, points=1, answer=0.16, check=numeric]{2cm}{0.17in}\quad \gradedblank[id=hw1c2, type=short, points=1, answer=(0.16-0.18)/sqrt(0.18*0.82/450), check=expression, display={\boldmath$(0.16-0.18)/0.0181$}]{6cm}{0.17in}\quad \gradedblank[id=hw1c3, type=short, points=2, answer=-1.10, check=numeric, tolerance=0.02]{2cm}{0.17in}}
        \end{enumerate}

  \newpage
  \item Seed germination test:
        \begin{enumerate}[label=\alph*.]
          \item \ansline{$H_0: p=0.85$; $H_a: p<0.85$ (one-sided). The researcher thinks the rate is \emph{lower} than claimed.}\gradedkeyline{\gradedblank[id=hw2a, type=rubric, points=2, display={$H_0\colon p = 0.85$, $H_a\colon p < 0.85$: the researcher suspects the rate is lower than claimed.}]{\linewidth+6pt}{\gradedlinesheight{1}}}
          \item \ansline{Random (assumed random planting --- met); Large Counts: $200(0.85)=170\ge10$, $200(0.15)=30\ge10$ --- met; 10\% met.}\gradedkeyline{\gradedblank[id=hw2b, type=rubric, points=3, display={Random sample of seeds; $200(0.85)=170 \ge 10$ and $200(0.15)=30 \ge 10$; $200 \le 10\%$ of all seeds. All met.}]{\linewidth+6pt}{\gradedlinesheight{2}}}
          \item $\hat p = $ \ans{$162/200=0.81$} \quad $z \approx$ \ans{$\dfrac{0.81-0.85}{\sqrt{0.85(0.15)/200}} = \dfrac{-0.04}{0.02525} \approx -1.58$}\gradedkey{\gradedblank[id=hw2c1, type=short, points=1, answer=0.81, check=numeric]{2cm}{0.17in}\quad \gradedblank[id=hw2c2, type=short, points=2, answer=-1.58, check=numeric, tolerance=0.02]{2cm}{0.17in}}
        \end{enumerate}

  \newpage
  \item Extracurricular participation test:
        \begin{enumerate}[label=\alph*.]
          \item \ansline{$H_0: p=0.70$; $H_a: p>0.70$ (one-sided --- parent group believes it increased)}\gradedkeyline{\gradedblank[id=hw3a, type=rubric, points=2, display={\bfseries $H_0\colon p = 0.70$; $H_a\colon p > 0.70$}]{\linewidth+6pt}{\gradedlinesheight{9}}}
          \item \ansline{Random (SRS --- met); Large Counts: $180(0.70)=126\ge10$, $180(0.30)=54\ge10$ --- met; 10\% met.}\gradedkeyline{\gradedblank[id=hw3b, type=rubric, points=3, display={Random; $180(0.70)=126 \ge 10$, $180(0.30)=54 \ge 10$; $180 \le 10\%$ of students. All met.}]{\linewidth+6pt}{\gradedlinesheight{1}}}
          \item $z \approx$ \ans{$\dfrac{0.75-0.70}{\sqrt{0.70(0.30)/180}} = \dfrac{0.05}{0.03416} \approx 1.46$}\gradedkey{\gradedblank[id=hw3c, type=short, points=2, answer=1.46, check=numeric, tolerance=0.02]{2cm}{0.17in}}
        \end{enumerate}

  \item \textbf{Reflection:} \ansline{If the researcher believed the county rate was \emph{higher}, $H_a$ would change to $p > 0.18$ (one-sided). The hypotheses and conditions remain the same; only the direction of $H_a$ changes, affecting which tail the $p$-value is computed from.}\gradedkeyline{\gradedblank[id=hw4, type=rubric, points=2, display={$H_a$ becomes $p > 0.18$ (one-sided); $H_0$ and the conditions stay the same.}]{\linewidth+6pt}{\gradedlinesheight{1}}}

\end{enumerate}
