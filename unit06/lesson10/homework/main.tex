\item \textbf{College Club Membership.} A sociologist wants to test whether the proportion
of first-generation college students who join a campus club differs from the proportion of
non-first-generation students who do. Sample 1 (first-gen): 48 of 160 joined.
Sample 2 (non-first-gen): 72 of 200 joined.

\begin{enumerate}[label=\alph*.]
  \item State $H_0$ and $H_a$. \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw1a, type=rubric, points=2, display={$H_0\colon p_1 = p_2$; $H_a\colon p_1 \ne p_2$ (true club-joining proportions, first-gen vs.\ not).}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{1}}
  \item Compute $\hat p_1$, $\hat p_2$, and $\hat p_c$. \writeline\\\gradedhide{\vspace{1pt}\noindent\textcolor{linegray}{\hrulefill}\gradedright{$\hat p_1 = $~\gradedblank[id=hw1b1, type=short, points=1, answer=0.300, check=numeric, tolerance=0.001]{0.7in}{0.17in}\quad $\hat p_2 = $~\gradedblank[id=hw1b2, type=short, points=1, answer=0.360, check=numeric, tolerance=0.001]{0.7in}{0.17in}\quad $\hat p_c = $~\gradedblank[id=hw1b3, type=short, points=1, answer=0.333, check=numeric, tolerance=0.001]{0.7in}{0.17in}}\vspace{3pt}}
  \item Verify all three conditions. \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw1c, type=rubric, points=3, display={Random samples; with $\hat p_c = 1/3$: $53.3, 106.7, 66.7, 133.3$ all $\ge 10$; each sample $\le 10\%$ of its group.}]{\linewidth+6pt}{\gradedlinesheight{2}}}\writelines{2}}
\end{enumerate}

\item \textbf{Defect Rate.} A manufacturer claims that its new production line has a lower
defect rate than the old one. Old line: 36 defects in 400 units. New line: 20 defects in 400 units.

\begin{enumerate}[label=\alph*.]
  \item Define $p_O$ and $p_N$, then state $H_0$ and $H_a$ (one-sided). \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2a, type=rubric, points=2, display={$p_O, p_N$: true defect rates of the old / new line. $H_0\colon p_O = p_N$; $H_a\colon p_O > p_N$.}]{\linewidth+6pt}{\gradedlinesheight{1}}}\writelines{1}}
  \item Compute $\hat p_c$. \writeline\\\gradedhide{\vspace{1pt}\noindent\textcolor{linegray}{\hrulefill}\gradedright{$\hat p_c = $~\gradedblank[id=hw2b, type=short, points=1, answer=0.070, check=numeric, tolerance=0.001]{0.8in}{0.17in}}\vspace{3pt}}
  \item Verify conditions. \gradedhide{\writeline\\\gradedhere{\gradedblank[id=hw2c, type=rubric, points=3, display={Random samples from each line; with $\hat p_c = 0.07$: $28, 372$ (each line) all $\ge 10$; each sample $\le 10\%$ of its line's units.}]{\linewidth+6pt}{\gradedlinesheight{2}}}\writelines{2}}
\end{enumerate}

\end{enumerate}
